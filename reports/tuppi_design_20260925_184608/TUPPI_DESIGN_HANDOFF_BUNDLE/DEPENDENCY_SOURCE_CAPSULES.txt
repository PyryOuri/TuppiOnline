"""Conversions between the authoritative TuppiProject engine and the native core.

`obs_from_player_observation` is the only path by which an engine observation
reaches a native encoder, and it reads only PlayerObservation fields, which the
engine already restricts to what the observing seat may legally know.
"""

from __future__ import annotations

from tuppi_engine import EnginePhase, PlayCardAction, ShowAction
from tuppi_rules import Contract
from tuppi_rules.cards import card_from_index, card_to_index

from .native import (ACT_EXCHANGE, ACT_SOOLI_ACCEPT, ACT_SOOLI_DECLINE, NOLO, PH_DONE, PH_PLAY, PH_SHOW,
                     PH_SOOLI_GIVE, PH_SOOLI_OFFER, PH_SOOLI_PLAY, PH_SOOLI_RETURN, RAMI, TcObs)

_PHASE = {EnginePhase.SHOWING: PH_SHOW, EnginePhase.TRICK_PLAY: PH_PLAY}
# project_target_v2 engines add the sooli phases; a v1-only engine lacks them.
for _name, _ph in (("SOOLI_OFFER", PH_SOOLI_OFFER), ("SOOLI_GIVE", PH_SOOLI_GIVE),
                   ("SOOLI_RETURN", PH_SOOLI_RETURN), ("SOOLI_PLAY", PH_SOOLI_PLAY)):
    if hasattr(EnginePhase, _name):
        _PHASE[getattr(EnginePhase, _name)] = _ph
try:
    from tuppi_engine import SooliDecisionAction, SooliExchangeAction
except ImportError:  # v1-only engine
    SooliDecisionAction = SooliExchangeAction = None
_CONTRACT = {Contract.NOLO: NOLO, Contract.RAMI: RAMI}
_CONTRACTS = (Contract.NOLO, Contract.RAMI)


def mask_of(cards) -> int:
    m = 0
    for card in cards:
        m |= 1 << card_to_index(card)
    return m


def action_id(action) -> int:
    if isinstance(action, PlayCardAction):
        return card_to_index(action.card)
    if isinstance(action, ShowAction):
        return 52 + 52 * _CONTRACT[action.declared_contract] + card_to_index(action.shown_card)
    if SooliDecisionAction is not None and isinstance(action, SooliDecisionAction):
        return ACT_SOOLI_ACCEPT if action.play_sooli else ACT_SOOLI_DECLINE
    if SooliExchangeAction is not None and isinstance(action, SooliExchangeAction):
        return ACT_EXCHANGE + card_to_index(action.card)
    raise TypeError(f"unknown action {action!r}")


def engine_action(action: int, seat: int):
    if action in (ACT_SOOLI_DECLINE, ACT_SOOLI_ACCEPT):
        return SooliDecisionAction(seat=seat, play_sooli=action == ACT_SOOLI_ACCEPT)
    if action >= ACT_EXCHANGE:
        return SooliExchangeAction(seat=seat, card=card_from_index(action - ACT_EXCHANGE))
    if action < 52:
        return PlayCardAction(seat=seat, card=card_from_index(action))
    contract, card = divmod(action - 52, 52)
    return ShowAction(seat=seat, shown_card=card_from_index(card),
                      declared_contract=_CONTRACTS[contract])


def obs_from_player_observation(observation) -> TcObs:
    o = TcObs()
    o.hand = mask_of(observation.own_hand)
    o.seat = observation.observing_seat
    o.dealer = observation.dealer
    o.phase = _PHASE.get(observation.phase, PH_DONE)
    shows = observation.show_actions
    # project_target_v3 showing: no bid is visible, only how many were taken.
    o.n_shows = len(shows) + getattr(observation, "bids_locked", 0)
    for i, show in enumerate(shows):
        o.show_seat[i] = show.seat
        o.show_card[i] = card_to_index(show.shown_card)
        o.show_contract[i] = _CONTRACT[show.declared_contract]
        o.show_verbal[i] = int(show.verbal_required)
    o.contract = -1 if observation.contract is None else _CONTRACT[observation.contract]
    o.declarer = -1 if observation.declarer is None else observation.declarer
    o.first_leader = 0 if observation.first_leader is None else observation.first_leader
    sooli = getattr(observation, "soolaaja", None) is not None and getattr(observation, "sooli_exchanged", False)
    if sooli:
        tricks, current = observation.sooli_tricks, observation.sooli_current_trick
    else:
        tricks, current = observation.completed_tricks, observation.current_trick
    n = 0
    for trick in tricks:
        for play in trick.plays:
            o.play_card[n] = card_to_index(play.card)
            o.play_seat[n] = play.seat
            n += 1
    for play in current:
        o.play_card[n] = card_to_index(play.card)
        o.play_seat[n] = play.seat
        n += 1
    o.n_played = n
    o.score[0] = observation.match_score.team0
    o.score[1] = observation.match_score.team1
    ruleset = getattr(observation, "ruleset_id", None)
    o.ruleset = {"project_target_v2": 2, "project_target_v3": 3}.get(getattr(ruleset, "value", None), 1)
    o.trick_size = 3 if sooli else 4
    decisions = getattr(observation, "sooli_decisions", ())
    o.n_decisions = len(decisions)
    for i, (seat, accept) in enumerate(decisions):
        o.decision_seat[i] = seat
        o.decision_accept[i] = int(accept)
    soolaaja = getattr(observation, "soolaaja", None)
    o.soolaaja = -1 if soolaaja is None else soolaaja
    given = getattr(observation, "sooli_given_card", None)
    returned = getattr(observation, "sooli_returned_card", None)
    o.known_given = -1 if given is None else card_to_index(given)
    o.known_returned = -1 if returned is None else card_to_index(returned)
    return o


=== E015 | AI | tuppi_ai/review.py | complete selected dependency ===
"""After-deal review: the player's own decisions, re-decided by the bot.

For each decision a seat took in a completed deal, the best agent's search is
run again from exactly that seat's observation at the time (never the full
deal), with the agent's own settings. Every legal action is priced on the same
sampled worlds, so the gap between the played action and the bot's choice has
a paired standard error. A decision is flagged as a clear mistake only if the
gap is at least ``min_cost`` match equity, at least ``z`` standard errors,
and both still hold when re-searched with ``confirm_worlds`` fresh worlds.

Equity is the match-win probability (the equity model of `tuppi_ai.equity`),
so a cost of 0.01 is one percentage point of the chance to win the match.

Not flagged, by design: forced decisions; the sooli return (the bot plays the
heuristic there, search measured no better); near-ties.

Calibration (`tools/review_calibrate.py`, docs/STATUS.md): at the defaults
(0.01, 2.5 SE, 1024 confirming worlds) the bot reviewing its own play flags
0.06 decisions per seat-deal and the incumbent 1.2. The estimated costs are
the search's: the incumbent's flagged costs add up to about six times its
measured deficit, so they rank mistakes better than they measure them.
"""

from __future__ import annotations

import ctypes as C
from dataclasses import dataclass

import numpy as np

from . import native
from .bridge import action_id, obs_from_player_observation

KIND_BID = "bid"


@dataclass(frozen=True)
class ReviewedDecision:
    decision_index: int
    kind: str
    trick_index: int | None
    played: int          # native action id
    recommended: int     # native action id (the bot's choice)
    cost: float          # recommended value - played value (>= 0 when flagged)
    standard_error: float
    flagged: bool


def decision_kind(observation) -> tuple[str, int | None]:
    """(kind, trick index) of a decision from the engine observation."""
    phase = observation.phase.value
    if phase == "SHOWING":
        return KIND_BID, None
    if phase == "SOOLI_OFFER":
        return "sooli_offer", None
    if phase == "SOOLI_GIVE":
        return "sooli_give", None
    if phase == "SOOLI_RETURN":
        return "sooli_return", None
    sooli = phase == "SOOLI_PLAY"
    tricks = observation.sooli_tricks if sooli else observation.completed_tricks
    current = observation.sooli_current_trick if sooli else observation.current_trick
    prefix = "sooli_" if sooli else ""
    if not current:
        return prefix + "lead", len(tricks)
    led = current[0].card.suit
    follows = any(card.suit is led for card in observation.own_hand)
    return prefix + ("follow" if follows else "discard"), len(tricks)


class DealReviewer:
    """Prices a seat's decisions with the best agent's search."""

    reviewer_id = "tuppi_ai.best_search_review.v1"

    def __init__(self, net, equity_table, *, worlds: int = 256, confirm_worlds: int = 1024,
                 dd_left: int = 6, min_cost: float = 0.01, z: float = 2.5, seed: int = 0,
                 sooli: dict | None = None):
        from .search import SOOLI_DEFAULTS

        self.mlp = native.Mlp(net)
        self.worlds, self.confirm_worlds, self.dd_left = worlds, confirm_worlds, dd_left
        self.min_cost, self.z = min_cost, z
        self.sooli = {**SOOLI_DEFAULTS, **(sooli or {})}
        self.env = native.Env(max(worlds, confirm_worlds) * 26, seed, equity_table, single_deal=True,
                              encoder=net.encoder)
        self.rng = C.c_uint64(seed * 7919 + 17)

    @classmethod
    def best(cls, **kwargs) -> "DealReviewer":
        from . import model as qmodel
        from .best import EQUITY, ROLLOUT, SOLVER_TAIL
        from .equity import Equity

        return cls(qmodel.load(ROLLOUT, "cpu"), Equity.load(EQUITY).table, dd_left=SOLVER_TAIL, **kwargs)

    def _values(self, obs: native.TcObs, ids: list[int], worlds: int) -> np.ndarray:
        # The same rollout settings as SearchAgent: sooli play inside rollouts
        # by the heuristics, except when pricing a sooli exchange or sooli card.
        cfg = self.sooli
        in_sooli = obs.phase in native.SOOLI_PHASES and obs.phase != native.PH_SOOLI_OFFER
        self.env.set_sooli(cfg["belief"], False, False if in_sooli else cfg["offer_play_heur"], cfg["lam"])
        return native.search_world_values(self.env, self.mlp, obs, ids, worlds, self.dd_left, self.rng)

    def _gap(self, per_world: np.ndarray, ids: list[int], played: int) -> tuple[int, float, float]:
        mean = per_world.mean(axis=0)
        best = int(np.argmax(mean))
        p = ids.index(played)
        diff = per_world[:, best] - per_world[:, p]
        se = float(diff.std(ddof=1) / np.sqrt(len(diff))) if len(diff) > 1 else float("inf")
        return best, float(diff.mean()), se

    def review_one(self, index: int, observation, action) -> ReviewedDecision | None:
        """None for a forced decision."""
        legal = observation.legal_actions
        if len(legal) < 2:
            return None
        kind, trick = decision_kind(observation)
        obs = obs_from_player_observation(observation)
        ids = [action_id(a) for a in legal]
        played = action_id(action)
        best, gap, se = self._gap(self._values(obs, ids, self.worlds), ids, played)
        recommended = ids[best]
        clear = kind != "sooli_return" and recommended != played and gap >= self.min_cost and gap >= self.z * se
        if clear:
            # Confirm on fresh, larger worlds with the recommendation held fixed:
            # choosing the best of many noisy means inflates the first gap, so
            # the reported cost is this unbiased paired estimate instead.
            pair = [recommended, played]
            per_world = self._values(obs, pair, self.confirm_worlds)
            diff = per_world[:, 0] - per_world[:, 1]
            gap = float(diff.mean())
            se = float(diff.std(ddof=1) / np.sqrt(len(diff)))
            clear = gap >= self.min_cost and gap >= self.z * se
        return ReviewedDecision(index, kind, trick, played, recommended, gap, se, clear)

    def review(self, decisions) -> list[ReviewedDecision]:
        """`decisions`: (engine observation, engine action) pairs of one seat, in order."""
        out = []
        for index, (observation, action) in enumerate(decisions):
            item = self.review_one(index, observation, action)
            if item is not None:
                out.append(item)
        return out


class ServerReviewer:
    """TuppiProject's ``tuppi_server.deal_review.DealReviewer`` backed by the best agent's search."""

    reviewer_id = DealReviewer.reviewer_id

    def __init__(self, reviewer: DealReviewer | None = None):
        self.reviewer = reviewer or DealReviewer.best()

    def review_decisions(self, decisions):
        from tuppi_server.deal_review import ReviewFinding

        from .bridge import engine_action

        findings = []
        for decision in decisions:
            item = self.reviewer.review_one(decision.decision_index, decision.observation, decision.action)
            if item is not None and item.flagged:
                findings.append(ReviewFinding(
                    decision_index=decision.decision_index,
                    recommended=engine_action(item.recommended, decision.action.seat),
                    cost=item.cost,
                    standard_error=item.standard_error,
                ))
        return findings

from __future__ import annotations

@dataclass(frozen=True, slots=True)
class MatchState:
    match_ref: str
    initial_score: MatchScore
    initial_dealer: int
    score: MatchScore
    dealer: int
    phase: EnginePhase
    active_deal: DealState | None
    completed_deals: tuple[ResearchDealResult, ...]
    public_events: tuple[PublicEvent, ...]
    ruleset_id: RulesetId = RulesetId.PROJECT_TARGET_V1

    def __post_init__(self) -> None:
        # Implicit revalidation of a state this engine just produced. Elided
        # inside ``trusted.trusted_transitions``; see that module.
        if not is_trusted():
            _validate_match_state(self)

    @property
    def current_actor(self) -> int | None:
        return None if self.active_deal is None else self.active_deal.current_actor

    @property
    def terminal(self) -> bool:
        return self.phase is EnginePhase.MATCH_COMPLETE


def create_match(
    *,
    initial_score: MatchScore,
    initial_dealer: int,
    match_ref: str,
    ruleset_id: RulesetId = RulesetId.PROJECT_TARGET_V1,
) -> MatchState:
    if not isinstance(initial_score, MatchScore):
        raise InvalidConstructionError("initial_score must be a MatchScore")
    try:
        checked_score = MatchScore(initial_score.team0, initial_score.team1)
    except ValueError as exc:
        raise InvalidConstructionError(f"invalid initial_score: {exc}") from exc
    if is_match_terminal(checked_score):
        raise InvalidConstructionError("match cannot start from a terminal score")
    try:
        dealer = validate_seat(initial_dealer)
    except ValueError as exc:
        raise InvalidConstructionError("initial_dealer must be an integer in 0..3") from exc
    if not is_valid_match_ref(match_ref):
        raise InvalidConstructionError("match_ref does not match the public schema")
    return MatchState(
        match_ref=match_ref,
        initial_score=checked_score,
        initial_dealer=dealer,
        score=checked_score,
        dealer=dealer,
        phase=EnginePhase.AWAITING_DEAL,
        active_deal=None,
        completed_deals=(),
        public_events=(),
        ruleset_id=ruleset_id,
    )


def begin_explicit_deal(state: MatchState, explicit_deal: ExplicitDeal) -> MatchState:
    validate_match_state(state)
    if state.terminal:
        raise TerminalStateError("completed match accepts no additional deals")
    if state.phase not in (EnginePhase.AWAITING_DEAL, EnginePhase.DEAL_COMPLETE):
        raise WrongPhaseError("cannot begin a deal while another deal is active")
    if not isinstance(explicit_deal, ExplicitDeal):
        raise InvalidConstructionError("explicit_deal must be an ExplicitDeal")
    if explicit_deal.dealer != state.dealer:
        raise DealSequenceError(
            f"explicit dealer {explicit_deal.dealer} does not match expected dealer {state.dealer}"
        )
    deal_number = len(state.completed_deals)
    deal_ref = f"{state.match_ref}/deal:{deal_number}"
    active = create_deal_state(
        explicit_deal,
        starting_score=state.score,
        match_ref=state.match_ref,
        deal_ref=deal_ref,
        deal_number=deal_number,
        event_start_index=len(state.public_events),
        ruleset_id=state.ruleset_id,
    )
    return MatchState(
        match_ref=state.match_ref,
        initial_score=state.initial_score,
        initial_dealer=state.initial_dealer,
        score=state.score,
        dealer=state.dealer,
        phase=EnginePhase.SHOWING,
        active_deal=active,
        completed_deals=state.completed_deals,
        public_events=state.public_events + active.events,
        ruleset_id=state.ruleset_id,
    )


def apply_match_action(state: MatchState, action: Action) -> MatchState:
    # The entry revalidation is implicit: the caller is handing back a state
    # this engine produced. Elided in a trusted block, where the type check is
    # kept so the error contract for a genuinely wrong argument survives.
    if is_trusted():
        if not isinstance(state, MatchState):
            raise StateInvariantError("state must be a MatchState")
    else:
        validate_match_state(state)
    if state.terminal:
        raise TerminalStateError("completed match accepts no actions")
    if state.phase is EnginePhase.DEAL_COMPLETE:
        raise TerminalStateError(
            "completed deal accepts no actions; begin the next explicit deal"
        )
    if state.phase not in ACTIVE_PHASES:
        raise WrongPhaseError("no active deal action is currently expected")
    if state.active_deal is None:
        raise StateInvariantError("active phase has no private deal state")
    old_event_count = len(state.active_deal.events)
    # validate_match_state above already revalidated this active deal, so the
    # dispatcher below repeats neither that work nor any of its own checks.
    active = _apply_action_validated(state.active_deal, action)
    new_public_events = state.public_events + active.events[old_event_count:]
    completed = state.completed_deals
    score = state.score
    dealer = state.dealer
    if active.complete:
        research_result = research_deal_result_from_state(active)
        completed = completed + (research_result,)
        if active.ending_score is None:
            raise StateInvariantError("completed deal has no ending score")
        score = active.ending_score
        dealer = active.dealer if active.next_dealer is None else active.next_dealer
    return MatchState(
        match_ref=state.match_ref,
        initial_score=state.initial_score,
        initial_dealer=state.initial_dealer,
        score=score,
        dealer=dealer,
        phase=active.phase,
        active_deal=active,
        completed_deals=completed,
        public_events=new_public_events,
        ruleset_id=state.ruleset_id,
    )


def match_observation_for(state: MatchState, observing_seat: int) -> PlayerObservation:
    if is_trusted():
        if not isinstance(state, MatchState):
            raise StateInvariantError("state must be a MatchState")
    else:
        validate_match_state(state)
    if state.active_deal is None:
        raise WrongPhaseError("there is no active deal to observe")
    # The match validation above revalidated the active deal; the seat check is
    # the same one ``observation_for`` performs, in the same order.
    return _observation_for_validated(
        state.active_deal, validate_seat(observing_seat)
    )


def play_active_deal(
    state: MatchState,
    agents: tuple[Agent, Agent, Agent, Agent],
    *,
    action_audit: Callable[[Action], None] | None = None,
) -> MatchState:
    """Run one finite deal, exposing only the current actor's observation."""

    validate_match_state(state)
    if state.phase not in ACTIVE_PHASES:
        raise WrongPhaseError("play_active_deal requires an active deal")
    if type(agents) is not tuple or len(agents) != 4:
        raise AgentProtocolError("runner requires exactly four agents")
    if len({id(agent) for agent in agents}) != 4:
        raise AgentProtocolError(
            "each seat requires a distinct agent object to preserve observation isolation"
        )
    if any(not callable(getattr(agent, "choose_action", None)) for agent in agents):
        raise AgentProtocolError("every agent must implement choose_action")

    current = state
    accepted_actions = 0
    while current.phase in ACTIVE_PHASES:
        actor = current.current_actor
        if actor is None:
            raise StateInvariantError("active deal has no current actor")
        observation = match_observation_for(current, actor)
        action = agents[actor].choose_action(observation)
        if action_audit is not None:
            action_audit(action)
        current = apply_match_action(current, action)
        accepted_actions += 1
        if accepted_actions > 60:
            raise StateInvariantError("deal exceeded 4 shows, 2 sooli answers, 2 exchanges and 52 plays")
    return current


def match_result_from_state(state: MatchState) -> MatchResult:
    validate_match_state(state)
    if not state.terminal:
        raise WrongPhaseError("MatchResult is available only after reaching the target")
    winner = match_winner(state.score)
    if winner is None:
        raise StateInvariantError("terminal match has no winner")
    return MatchResult(
        match_ref=state.match_ref,
        initial_score=state.initial_score,
        initial_dealer=state.initial_dealer,
        final_score=state.score,
        winner_team=winner,
        deals=tuple(item.public_result for item in state.completed_deals),
        events=state.public_events,
    )


def validate_match_state(state: MatchState) -> MatchState:
    if not isinstance(state, MatchState):
        raise StateInvariantError("state must be a MatchState")
    _validate_match_state(state)
    return state


def _validate_match_state(state: MatchState) -> None:
    if state.ruleset_id not in SUPPORTED_RULESETS:
        raise StateInvariantError("match supports project_target_v1 and project_target_v2")
    if state.active_deal is not None and state.active_deal.ruleset_id is not state.ruleset_id:
        raise StateInvariantError("active deal ruleset disagrees with the match")
    if not is_valid_match_ref(state.match_ref):
        raise StateInvariantError("match_ref does not match the public schema")
    if not isinstance(state.initial_score, MatchScore):
        raise StateInvariantError("initial score must be nonterminal")
    try:
        checked_initial_score = MatchScore(
            state.initial_score.team0, state.initial_score.team1
        )
        checked_score = MatchScore(state.score.team0, state.score.team1)
    except (AttributeError, ValueError) as exc:
        raise StateInvariantError("match score contains invalid fields") from exc
    if checked_initial_score != state.initial_score or is_match_terminal(
        checked_initial_score
    ):
        raise StateInvariantError("initial score must be valid and nonterminal")
    for label, value in (("initial_dealer", state.initial_dealer), ("dealer", state.dealer)):
        if type(value) is not int or value not in range(4):
            raise StateInvariantError(f"{label} must be an integer in 0..3")
    if (
        not isinstance(state.score, MatchScore)
        or checked_score != state.score
        or not isinstance(state.phase, EnginePhase)
    ):
        raise StateInvariantError("match score or phase is malformed")
    if type(state.completed_deals) is not tuple or any(
        not isinstance(result, ResearchDealResult) for result in state.completed_deals
    ):
        raise StateInvariantError("completed_deals must be immutable research results")
    if type(state.public_events) is not tuple or any(
        not isinstance(event, PublicEvent) for event in state.public_events
    ):
        raise StateInvariantError("public_events must be an immutable event tuple")
    if tuple(event.event_index for event in state.public_events) != tuple(
        range(len(state.public_events))
    ):
        raise StateInvariantError("match event indices must be contiguous from zero")
    if any(event.match_ref != state.match_ref for event in state.public_events):
        raise StateInvariantError("event match_ref disagrees with match state")

    for deal_number, research in enumerate(state.completed_deals):
        public = research.public_result
        if public.deal_number != deal_number or public.match_ref != state.match_ref:
            raise StateInvariantError("completed deal order or match reference is invalid")
        if deal_number == 0:
            if public.dealer != state.initial_dealer or public.score_before != state.initial_score:
                raise StateInvariantError("first result disagrees with initial match state")
        else:
            previous = state.completed_deals[deal_number - 1].public_result
            if previous.match_terminal:
                raise StateInvariantError("completed result exists after terminal match")
            if public.dealer != previous.next_dealer or public.score_before != previous.score_after:
                raise StateInvariantError("completed deal sequence is discontinuous")

    completed_events = tuple(
        event
        for research in state.completed_deals
        for event in research.public_result.events
    )
    if state.phase is EnginePhase.AWAITING_DEAL:
        if state.active_deal is not None or state.completed_deals or state.public_events:
            raise StateInvariantError("AWAITING_DEAL is only the pristine match state")

        raise StateInvariantError("active deal belongs to a different match")
    if state.active_deal.deal_number != len(state.completed_deals) - (

    if state.phase is not state.active_deal.phase:

        if is_match_terminal(state.score):

"""Exact single-suit combinatorics: how the rest of a suit lies, and what it costs.

Everything in this module is exact rational arithmetic over the real deal space.
There is no sampling and no approximation, so a returned probability is the
probability, not an estimate of one.

Two questions are answered here, and they are different:

*How does the suit lie?*
    :func:`split_distribution` and :func:`pattern_distribution` give the exact
    multivariate-hypergeometric law of the outstanding cards across the other
    three seats. This is pure dealing combinatorics and involves no play at all.

*What does the holding take?*
    :func:`trick_distribution` plays the suit out exactly, over every possible
    layout of the outstanding ranks, weighted by that layout's true probability.

The second question cannot be answered without saying how the cards are played,
so the play rule is an explicit, named parameter rather than a hidden default.
Two rules are offered, and both are exactly computable:

``SuitPolicy.LOWEST``
    Every seat contributes its lowest remaining card of the suit.

``SuitPolicy.HIGHEST``
    Every seat contributes its highest remaining card.

**Neither rule is a proxy for a contract, and the obvious mapping is backwards.**
Phase D measured real follower play: under RAMI 87.6% of cards played with a
genuine choice were the seat's *lowest* of the suit, and under NOLO 51.0% were
its *highest*. Card economy explains it - in RAMI you spend the cheapest card
that does the job and never overtake a partner who is already winning (97.8%
lowest in that case), while in NOLO you unload dangerous high cards onto a trick
your side is taking anyway (76.3% highest when the partner is winning).

**Nor do the two rules bracket the answer.** Playing highest is not an upper
bound: K-7-2 over three leads takes 0.79 tricks under LOWEST and exactly zero
under HIGHEST, because leading the king into the outstanding ace throws it away.
See ``test_competing_is_not_dominant_over_ducking``.

Treat the two rules as two exactly-solvable reference behaviours, not as bounds
and not as contract proxies. For what a descriptor is actually worth under each
contract, measure it - ``docs/valuation/phase_c_results.md`` does.

Because a trick is won by the highest card of the led suit and every holder must
follow, the winner under a fixed rule does not depend on the order the seats play
in. That is what makes an exact answer available without a game tree.

The number of times the suit is led is exogenous here: it is an input, not a
prediction. :mod:`tuppi_valuation.corpus` supplies the realistic distribution of
lead counts measured from actual play.
"""

from __future__ import annotations

from collections.abc import Iterator, Sequence
from dataclasses import dataclass, field
from enum import StrEnum
from fractions import Fraction
from functools import lru_cache
import itertools

from tuppi_exact.counting import multinomial
from tuppi_rules.cards import RANK_ORDER

from .errors import CombinatoricsError
from .holdings import Holding, validate_ranks


SUIT_ANALYSIS_SCHEMA_VERSION = "tuppi.valuation.suit_analysis.v1"

#: Seats other than the one holding the analysed suit.
OPPONENT_COUNT = 3
#: Cards each of those seats holds in total, across all suits.
CARDS_PER_HAND = 13
#: Cards of one suit in the deck.
SUIT_SIZE = len(RANK_ORDER)
#: Cards not in the analysed seat's hand.
UNSEEN_CARDS = CARDS_PER_HAND * OPPONENT_COUNT

#: Refuse an enumeration larger than this many layouts. 3^13 is 1.59M, which is
#: the worst case (a void), and is allowed; the guard exists so a caller that
#: passes a nonsense length gets an error instead of a hang.
MAX_LAYOUTS = 3 ** SUIT_SIZE


class SuitPolicy(StrEnum):
    """How each seat chooses which card of the suit to contribute."""

    LOWEST = "lowest"
    HIGHEST = "highest"


def _check_length(own_length: int) -> int:
    if type(own_length) is not int or own_length not in range(SUIT_SIZE + 1):
        raise CombinatoricsError("own_length must be an integer in 0..13")
    return own_length


def _total_deals() -> int:
    """Ways to deal the 39 unseen cards into three labelled 13-card hands."""

    return multinomial(UNSEEN_CARDS, (CARDS_PER_HAND,) * OPPONENT_COUNT)


def _layout_weight(counts: Sequence[int], outstanding: int) -> int:
    """Ways to complete a deal in which the opponents hold *counts* of the suit.

    The suit cards are already placed, so this counts only the ways to fill the
    remaining space with the non-suit unseen cards.
    """

    remainder = UNSEEN_CARDS - outstanding
    parts = tuple(CARDS_PER_HAND - count for count in counts)
    if any(part < 0 for part in parts):
        return 0
    return multinomial(remainder, parts)


@lru_cache(maxsize=None)
def split_distribution(own_length: int) -> dict[tuple[int, int, int], Fraction]:
    """Exact law of the ordered split of the outstanding cards.

    Returns ``{(a, b, c): probability}`` where *a*, *b*, *c* are the counts held
    by the three other seats in a fixed seat order. Probabilities sum to 1.
    """

    length = _check_length(own_length)
    outstanding = SUIT_SIZE - length
    total = _total_deals()
    distribution: dict[tuple[int, int, int], Fraction] = {}
    for counts in itertools.product(range(outstanding + 1), repeat=OPPONENT_COUNT):
        if sum(counts) != outstanding:
            continue
        ways = multinomial(outstanding, counts) * _layout_weight(counts, outstanding)
        if ways:
            distribution[counts] = Fraction(ways, total)
    if sum(distribution.values()) != 1:
        raise CombinatoricsError("ordered split distribution does not sum to one")
    return distribution


@lru_cache(maxsize=None)
def pattern_distribution(own_length: int) -> dict[tuple[int, ...], Fraction]:
    """Exact law of the *unordered* split, e.g. ``(3, 1, 1)``.

    This is the form a player actually reasons in: not "which opponent" but
    "how evenly". Keys are counts sorted descending.
    """

    patterns: dict[tuple[int, ...], Fraction] = {}
    for counts, probability in split_distribution(own_length).items():
        key = tuple(sorted(counts, reverse=True))
        patterns[key] = patterns.get(key, Fraction(0)) + probability
    return dict(sorted(patterns.items(), key=lambda item: -item[1]))


def opponent_count_distribution(own_length: int) -> dict[int, Fraction]:
    """Exact law of how many of the suit one *named* opponent holds.

    The plain hypergeometric marginal: 13 of the 39 unseen cards go to that
    seat, and *outstanding* of those 39 are in the suit.
    """

    length = _check_length(own_length)
    outstanding = SUIT_SIZE - length
    marginals: dict[int, Fraction] = {}
    for counts, probability in split_distribution(length).items():
        marginals[counts[0]] = marginals.get(counts[0], Fraction(0)) + probability
    if outstanding and sum(marginals.values()) != 1:
        raise CombinatoricsError("marginal distribution does not sum to one")
    return dict(sorted(marginals.items()))


def any_void_probability(own_length: int) -> Fraction:
    """Exact probability that at least one other seat holds none of the suit."""

    return sum(
        (
            probability
            for counts, probability in split_distribution(own_length).items()
            if 0 in counts
        ),
        Fraction(0),
    )


@lru_cache(maxsize=None)
def rounds_survived_distribution(own_length: int) -> dict[int, Fraction]:
    """Exact law of M, the number of rounds the suit survives among opponents.

    M is the largest number of the suit held by any one other seat. After M
    rounds nobody else can follow, so every further lead in the suit is won by
    whoever still holds one.
    """

    law: dict[int, Fraction] = {}
    for counts, probability in split_distribution(own_length).items():
        rounds = max(counts)
        law[rounds] = law.get(rounds, Fraction(0)) + probability
    return dict(sorted(law.items()))


def bottom_holding_trick_distribution(
    own_length: int, *, times_led: int
) -> dict[int, Fraction]:
    """Exact trick law for a holding of the *lowest* cards in its suit.

    A closed form, with no layout enumeration and **no play rule at all**:

        tricks = max(0, min(times_led, own_length) - M)

    Holding only the bottom cards, you can never beat anybody while an opponent
    still holds the suit, and you necessarily win once nobody does - so the play
    rule cannot matter, and the answer is pure dealing combinatorics. Verified
    against the full enumeration under both policies in ``test_suits.py``.

    This is the NOLO-relevant shape: it is the *best case* for taking no tricks,
    so its P(0) is an upper bound for any holding of that length.
    """

    length = _check_length(own_length)
    if type(times_led) is not int or times_led < 0:
        raise CombinatoricsError("times_led must be a nonnegative integer")
    distribution: dict[int, Fraction] = {}
    for rounds, probability in rounds_survived_distribution(length).items():
        wins = max(0, min(times_led, length) - rounds)
        distribution[wins] = distribution.get(wins, Fraction(0)) + probability
    return dict(sorted(distribution.items()))


def probability_stuck_after(own_length: int, rounds: int) -> Fraction:
    """P(no opponent still holds the suit after *rounds* rounds) = P(M <= rounds)."""

    if type(rounds) is not int or rounds < 0:
        raise CombinatoricsError("rounds must be a nonnegative integer")
    return sum(
        (
            probability
            for survived, probability in rounds_survived_distribution(
                own_length
            ).items()
            if survived <= rounds
        ),
        Fraction(0),
    )


def general_split_distribution(
    *,
    outstanding: int,
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool] = (False, False, False),
) -> dict[tuple[int, int, int], Fraction]:
    """Exact split law of *outstanding* cards across three hands of known size.

    The mid-deal generalisation of :func:`split_distribution`, which assumes a
    freshly dealt suit: thirteen cards outstanding across three thirteen-card
    hands. Part-way through a deal neither holds, and applying the opening law
    to a late position is simply wrong - it was measured to cancel out the whole
    benefit of a rule built on it.

    Three things are known publicly part-way through a deal and all three are
    used here: how many of the suit are still unaccounted for, how many cards
    each opponent still holds, and which opponents have *proved* void by failing
    to follow. A proven void is the strongest constraint available and collapses
    the law sharply, so it is taken rather than approximated.
    """

    if type(outstanding) is not int or outstanding < 0:
        raise CombinatoricsError("outstanding must be a nonnegative integer")
    if len(hand_sizes) != OPPONENT_COUNT or len(void) != OPPONENT_COUNT:
        raise CombinatoricsError("three hand sizes and three void flags are required")
    if any(type(size) is not int or size < 0 for size in hand_sizes):
        raise CombinatoricsError("hand sizes must be nonnegative integers")
    unseen = sum(hand_sizes)
    if outstanding > unseen:
        raise CombinatoricsError("more of the suit is outstanding than is unseen")

    total = multinomial(unseen, tuple(hand_sizes))
    if total == 0:
        raise CombinatoricsError("no legal deal of the unseen cards exists")

    distribution: dict[tuple[int, int, int], Fraction] = {}
    for counts in itertools.product(
        *(range(0, min(outstanding, size) + 1) for size in hand_sizes)
    ):
        if sum(counts) != outstanding:
            continue
        if any(flag and count for flag, count in zip(void, counts)):
            continue
        remainder = tuple(
            size - count for size, count in zip(hand_sizes, counts)
        )
        ways = multinomial(outstanding, counts) * multinomial(
            unseen - outstanding, remainder
        )
        if ways:
            distribution[counts] = Fraction(ways, total)
    mass = sum(distribution.values(), Fraction(0))
    if mass == 0:
        raise CombinatoricsError(
            "the stated voids and hand sizes admit no placement of the suit"
        )
    # Renormalise: the void flags condition the law on public evidence.
    return {counts: value / mass for counts, value in distribution.items()}


def general_rounds_survived(
    *,
    outstanding: int,
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool] = (False, False, False),
) -> dict[int, Fraction]:
    """Law of M mid-deal: rounds the suit survives among the opponents."""

    law: dict[int, Fraction] = {}
    for counts, probability in general_split_distribution(
        outstanding=outstanding, hand_sizes=hand_sizes, void=void
    ).items():
        rounds = max(counts)
        law[rounds] = law.get(rounds, Fraction(0)) + probability
    return dict(sorted(law.items()))


def expected_length_winners(
    *,
    own_length: int,
    outstanding: int,
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool] = (False, False, False),
) -> Fraction:
    """Tricks this holding takes unopposed once nobody else can follow.

    ``E[max(0, own_length - M)]``. This is the *ceiling*: collecting those
    tricks also requires keeping the lead, which this does not model.
    """

    if type(own_length) is not int or own_length < 0:
        raise CombinatoricsError("own_length must be a nonnegative integer")
    law = general_rounds_survived(
        outstanding=outstanding, hand_sizes=hand_sizes, void=void
    )
    return sum(
        (Fraction(max(0, own_length - rounds)) * probability
         for rounds, probability in law.items()),
        Fraction(0),
    )


@lru_cache(maxsize=None)
def expected_suit_tricks(
    *,
    own_length: int,
    top_sequence: int,
    outstanding: int,
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool] = (False, False, False),
) -> Fraction:
    """Expected tricks from playing a suit out, under the law that applies now.

    Two sources, taken inside one expectation over M:

    * ``min(top_sequence, M)`` - the top cards win while somebody can still
      follow, but never more rounds than the suit survives;
    * ``max(0, own_length - M)`` - every later round is unopposed.

    Taking the expectation *inside* matters. ``min`` is concave, so
    ``min(top_sequence, E[M])`` overstates the honour term by Jensen's
    inequality - biasing the score back toward honour-heavy suits, which is the
    bias a length-aware score exists to remove.

    Still a model, not the exact trick law: it assumes the suit is played out and
    that the top cards actually win the rounds they are spent on. The exact
    figure is :func:`trick_curve`, which is too slow to run inside play.
    """

    if type(top_sequence) is not int or top_sequence < 0:
        raise CombinatoricsError("top_sequence must be a nonnegative integer")
    if type(own_length) is not int or own_length < 0:
        raise CombinatoricsError("own_length must be a nonnegative integer")
    if top_sequence > own_length:
        raise CombinatoricsError("top_sequence cannot exceed own_length")
    law = general_rounds_survived(
        outstanding=outstanding, hand_sizes=hand_sizes, void=void
    )
    return sum(
        (
            probability * Fraction(
                min(top_sequence, rounds) + max(0, own_length - rounds)
            )
            for rounds, probability in law.items()
        ),
        Fraction(0),
    )


def layouts(
    own_ranks: Sequence[int],
) -> Iterator[tuple[tuple[tuple[int, ...], ...], Fraction]]:
    """Yield every exact layout of the outstanding ranks, with its probability.

    A layout is a 3-tuple of descending rank tuples, one per other seat. The
    yielded probabilities sum to 1. Enumeration is over 3**outstanding
    assignments, which is 243 for an eight-card holding and 1.59M for a void.
    """

    ranks = validate_ranks(own_ranks)
    held = set(ranks)
    outstanding_ranks = tuple(
        rank.value for rank in reversed(RANK_ORDER) if rank.value not in held
    )
    outstanding = len(outstanding_ranks)
    if 3 ** outstanding > MAX_LAYOUTS:
        raise CombinatoricsError("layout enumeration exceeds the exact budget")
    total = _total_deals()
    for assignment in itertools.product(range(OPPONENT_COUNT), repeat=outstanding):
        buckets: tuple[list[int], ...] = ([], [], [])
        for rank, owner in zip(outstanding_ranks, assignment):
            buckets[owner].append(rank)
        counts = tuple(len(bucket) for bucket in buckets)
        ways = _layout_weight(counts, outstanding)
        if not ways:
            continue
        yield tuple(tuple(bucket) for bucket in buckets), Fraction(ways, total)


def play_out(
    own_ranks: Sequence[int],
    opponent_ranks: Sequence[Sequence[int]],
    *,
    times_led: int,
    policy: SuitPolicy,
) -> tuple[int, ...]:
    """Play the suit out *times_led* times and return tricks won per seat.

    Index 0 is the analysed seat. Every seat that still holds the suit follows;
    a seat that has run out contributes nothing and cannot win. The trick goes
    to the highest card contributed, which under a fixed policy makes the result
    independent of the order the seats play in.
    """

    if type(times_led) is not int or times_led < 0:
        raise CombinatoricsError("times_led must be a nonnegative integer")
    if not isinstance(policy, SuitPolicy):
        raise CombinatoricsError("policy must be a SuitPolicy value")
    if len(opponent_ranks) != OPPONENT_COUNT:
        raise CombinatoricsError("exactly three opponent holdings are required")

    hands = [sorted(own_ranks)] + [sorted(hand) for hand in opponent_ranks]
    wins = [0, 0, 0, 0]
    for _ in range(times_led):
        contributed: dict[int, int] = {}
        for seat, hand in enumerate(hands):
            if not hand:
                continue
            contributed[seat] = hand.pop(0 if policy is SuitPolicy.LOWEST else -1)
        if not contributed:
            break
        winner = max(contributed, key=lambda seat: contributed[seat])
        wins[winner] += 1
    return tuple(wins)


def play_out_curve(
    own_ranks: Sequence[int],
    opponent_ranks: Sequence[Sequence[int]],
    *,
    max_leads: int,
    policy: SuitPolicy,
) -> tuple[int, ...]:
    """Cumulative tricks the analysed seat has won after each of *max_leads*.

    The same play-out as :func:`play_out`, reported after every lead instead of
    only at the end. One enumeration then serves every lead count, which is what
    makes a full sweep over all holdings affordable.
    """

    if type(max_leads) is not int or max_leads < 0:
        raise CombinatoricsError("max_leads must be a nonnegative integer")
    if not isinstance(policy, SuitPolicy):
        raise CombinatoricsError("policy must be a SuitPolicy value")
    if len(opponent_ranks) != OPPONENT_COUNT:
        raise CombinatoricsError("exactly three opponent holdings are required")

    hands = [sorted(own_ranks)] + [sorted(hand) for hand in opponent_ranks]
    running = 0
    curve: list[int] = []
    for _ in range(max_leads):
        contributed: dict[int, int] = {}
        for seat, hand in enumerate(hands):
            if hand:
                contributed[seat] = hand.pop(
                    0 if policy is SuitPolicy.LOWEST else -1
                )
        if contributed and max(contributed, key=lambda s: contributed[s]) == 0:
            running += 1
        curve.append(running)
    return tuple(curve)


def trick_curve(
    own_ranks: Sequence[int],
    *,
    max_leads: int,
    policy: SuitPolicy = SuitPolicy.LOWEST,
) -> dict[int, "SuitTrickAnalysis"]:
    """Exact trick law at every lead count from 1 to *max_leads*, in one pass."""

    ranks = validate_ranks(own_ranks)
    accumulated: list[dict[int, Fraction]] = [
        {} for _ in range(max_leads)
    ]
    for opponents, probability in layouts(ranks):
        curve = play_out_curve(
            ranks, opponents, max_leads=max_leads, policy=policy
        )
        for index, wins in enumerate(curve):
            bucket = accumulated[index]
            bucket[wins] = bucket.get(wins, Fraction(0)) + probability
    results: dict[int, SuitTrickAnalysis] = {}
    for index, distribution in enumerate(accumulated):
        if sum(distribution.values()) != 1:
            raise CombinatoricsError("trick curve does not sum to one")
        results[index + 1] = SuitTrickAnalysis(
            holding=Holding(suit=None, ranks=ranks).code,
            length=len(ranks),
            times_led=index + 1,
            policy=policy,
            distribution=dict(sorted(distribution.items())),
        )
    return results


def ducking_round_probabilities(
    own_ranks: Sequence[int], *, max_leads: int
) -> tuple[Fraction, ...]:
    """P(win round i) for i = 1..max_leads, when every seat ducks.

    Under ``SuitPolicy.LOWEST`` round *i* is contested by each seat's *i*-th
    lowest card of the suit and by nothing else, so the rounds are independent
    and

        E[tricks] = sum of these probabilities

    exactly. Two consequences worth knowing:

    * A holding's value under ducking is **additive across its own cards**:
      changing one card moves exactly one term. That is why the honour and the
      spot contributions to a holding add with no interaction at all.
    * When the suit is led *k* times, only your *lowest k* cards can matter.
      High cards you never reach are worth nothing, which is the reason
      high-card points measure almost the wrong thing for NOLO.

    Verified against the full enumeration in ``test_suits.py``.
    """

    ranks = validate_ranks(own_ranks)
    if type(max_leads) is not int or max_leads < 0:
        raise CombinatoricsError("max_leads must be a nonnegative integer")
    mine = sorted(ranks)
    rounds = min(max_leads, len(mine))
    totals = [Fraction(0)] * rounds
    for opponents, probability in layouts(ranks):
        hands = [sorted(hand) for hand in opponents]
        for index in range(rounds):
            others = [hand[index] for hand in hands if len(hand) > index]
            if not others or mine[index] > max(others):
                totals[index] += probability
    return tuple(totals) + (Fraction(0),) * (max_leads - rounds)


@dataclass(frozen=True, slots=True)
class SuitTrickAnalysis:
    """The exact trick law for one holding, one lead count, one play rule."""

    holding: str
    length: int
    times_led: int
    policy: SuitPolicy
    distribution: dict[int, Fraction]
    schema_version: str = field(
        default=SUIT_ANALYSIS_SCHEMA_VERSION, init=False
    )

    @property
    def expected_tricks(self) -> Fraction:
        return sum(
            (Fraction(tricks) * probability
             for tricks, probability in self.distribution.items()),
            Fraction(0),
        )

    @property
    def probability_none(self) -> Fraction:
        """P(this holding wins no trick in its own suit). The NOLO question."""

        return self.distribution.get(0, Fraction(0))

    def probability_at_least(self, tricks: int) -> Fraction:
        return sum(
            (probability
             for won, probability in self.distribution.items() if won >= tricks),
            Fraction(0),
        )

    @property
    def probability_all(self) -> Fraction:
        """P(this holding wins every trick led in its suit). The RAMI question."""

        return self.distribution.get(min(self.times_led, self.length), Fraction(0))

    def summary(self) -> dict[str, object]:
        return {
            "holding": self.holding,
            "length": self.length,
            "times_led": self.times_led,
            "policy": str(self.policy),
            "expected_tricks": float(self.expected_tricks),
            "probability_none": float(self.probability_none),
            "probability_all": float(self.probability_all),
            "distribution": {
                tricks: float(probability)
                for tricks, probability in sorted(self.distribution.items())
            },
        }


def trick_distribution(
    own_ranks: Sequence[int],
    *,
    times_led: int,
    policy: SuitPolicy = SuitPolicy.LOWEST,
) -> SuitTrickAnalysis:
    """Exact distribution of tricks this holding wins in its own suit.

    Averaged over every layout of the outstanding cards, weighted by that
    layout's exact probability.
    """

    ranks = validate_ranks(own_ranks)
    distribution: dict[int, Fraction] = {}
    for opponents, probability in layouts(ranks):
        wins = play_out(
            ranks, opponents, times_led=times_led, policy=policy
        )[0]
        distribution[wins] = distribution.get(wins, Fraction(0)) + probability
    total = sum(distribution.values(), Fraction(0))
    if total != 1:
        raise CombinatoricsError("trick distribution does not sum to one")
    return SuitTrickAnalysis(
        holding=Holding(suit=None, ranks=ranks).code,
        length=len(ranks),
        times_led=times_led,
        policy=policy,
        distribution=dict(sorted(distribution.items())),
    )


def compare_holdings(
    codes: Sequence[Sequence[int]],
    *,
    times_led: int,
    policy: SuitPolicy = SuitPolicy.LOWEST,
) -> tuple[SuitTrickAnalysis, ...]:
    """Analyse several holdings under identical assumptions, for comparison."""

    return tuple(
        trick_distribution(ranks, times_led=times_led, policy=policy)
        for ranks in codes

# ------------------------------------------------- conditioned ducking law
#
# The functions above take the *opening* prior: thirteen cards outstanding
# across three thirteen-card hands. Applying that to a mid-deal position was
# measured to cancel out the entire benefit of a rule built on it
# (docs/valuation/handoff.md 4.2), so the ducking law below conditions on what
# is publicly known.
#
# What it conditions on is deliberately narrow. Cards seen are cards gone, and a
# proven void is a certainty. Which *rank* an opponent chose to play is not used:
# read as an exact order statistic it looks highly informative, but under the
# honest reading ``min <= c`` only 1-5% of that signal survives, and the rest was
# measuring the baseline's determinism (handoff.md 3.1).


def _capped_compositions(
    total: int, caps: Sequence[int]
) -> Iterator[tuple[int, ...]]:
    """Yield every way to split *total* into three parts within *caps*."""

    first_cap, second_cap, third_cap = caps
    for first in range(min(total, first_cap) + 1):
        remaining = total - first
        low = max(0, remaining - third_cap)
        for second in range(low, min(remaining, second_cap) + 1):
            third = remaining - second
            if third <= third_cap:
                yield first, second, third


def _conditioned_outstanding(
    own_ranks: Sequence[int], seen_ranks: Sequence[int]
) -> tuple[int, ...]:
    """Ranks of the suit that are neither held nor already played, descending."""

    held = set(validate_ranks(own_ranks))
    seen = set(validate_ranks(seen_ranks)) if len(seen_ranks) else set()
    if held & seen:
        raise CombinatoricsError("a held rank cannot also have been seen")
    return tuple(
        rank.value
        for rank in reversed(RANK_ORDER)
        if rank.value not in held and rank.value not in seen
    )


def _check_conditioning(
    outstanding: int, hand_sizes: Sequence[int], void: Sequence[bool]
) -> int:
    if len(hand_sizes) != OPPONENT_COUNT or len(void) != OPPONENT_COUNT:
        raise CombinatoricsError("three hand sizes and three void flags are required")
    if any(type(size) is not int or size < 0 for size in hand_sizes):
        raise CombinatoricsError("hand sizes must be nonnegative integers")
    if any(type(flag) is not bool for flag in void):
        raise CombinatoricsError("void flags must be booleans")
    unseen = sum(hand_sizes)
    if outstanding > unseen:
        raise CombinatoricsError("more of the suit is outstanding than is unseen")
    capacity = sum(
        0 if flag else size for flag, size in zip(void, hand_sizes)
    )
    if outstanding > capacity:
        raise CombinatoricsError(
            "the stated voids and hand sizes admit no placement of the suit"
        )
    return unseen


def conditional_layouts(
    own_ranks: Sequence[int],
    *,
    seen_ranks: Sequence[int] = (),
    hand_sizes: tuple[int, int, int] = (CARDS_PER_HAND,) * OPPONENT_COUNT,
    void: tuple[bool, bool, bool] = (False,) * OPPONENT_COUNT,
) -> Iterator[tuple[tuple[tuple[int, ...], ...], Fraction]]:
    """Yield every layout of the *unseen* outstanding ranks, with probability.

    The mid-deal generalisation of :func:`layouts`. Cards in *seen_ranks* are
    removed from circulation, hand sizes may differ (they do part-way through a
    trick, where seats that have already played hold one card fewer), and a
    proven void forbids any placement in that hand. Probabilities are
    renormalised over the layouts the evidence permits and sum to 1.

    This enumerates ``3**outstanding`` rank assignments and is the **reference**
    implementation: correct, and far too slow to consult inside a decision. Use
    it to verify a faster law, which is what
    ``test_conditioned_ducking_matches_the_reference_enumeration`` does.
    """

    outstanding_ranks = _conditioned_outstanding(own_ranks, seen_ranks)
    outstanding = len(outstanding_ranks)
    unseen = _check_conditioning(outstanding, hand_sizes, void)
    if 3 ** outstanding > MAX_LAYOUTS:
        raise CombinatoricsError("layout enumeration exceeds the exact budget")

    placements: list[tuple[tuple[tuple[int, ...], ...], int]] = []
    total = 0
    for assignment in itertools.product(range(OPPONENT_COUNT), repeat=outstanding):
        buckets: tuple[list[int], ...] = ([], [], [])
        for rank, owner in zip(outstanding_ranks, assignment):
            buckets[owner].append(rank)
        counts = tuple(len(bucket) for bucket in buckets)
        if any(flag and count for flag, count in zip(void, counts)):
            continue
        remainder = tuple(
            size - count for size, count in zip(hand_sizes, counts)
        )
        if any(part < 0 for part in remainder):
            continue
        ways = multinomial(unseen - outstanding, remainder)
        if not ways:
            continue
        placements.append((tuple(tuple(bucket) for bucket in buckets), ways))
        total += ways
    if total == 0:
        raise CombinatoricsError(
            "the stated voids and hand sizes admit no placement of the suit"
        )
    for layout, ways in placements:
        yield layout, Fraction(ways, total)


def conditional_ducking_round_probabilities(
    own_ranks: Sequence[int],
    *,
    max_leads: int,
    seen_ranks: Sequence[int] = (),
    hand_sizes: tuple[int, int, int] = (CARDS_PER_HAND,) * OPPONENT_COUNT,
    void: tuple[bool, bool, bool] = (False,) * OPPONENT_COUNT,
) -> tuple[Fraction, ...]:
    """P(win round *i*) for i = 1..max_leads when every seat ducks, conditioned.

    The mid-deal counterpart of :func:`ducking_round_probabilities`, and exactly
    equal to it when nothing has been seen, no seat is void and every hand still
    holds thirteen cards.

    It is fast because it never enumerates rank assignments. Under
    ``SuitPolicy.LOWEST`` round *i* is contested by each seat's *i*-th lowest
    card, and an opponent's *i*-th lowest card lies **below** a given card
    exactly when that opponent holds *i* or more cards below it. So the round
    turns only on how the outstanding ranks split into those below the card and
    those above it, never on which particular ranks they are. That collapses
    ``3**outstanding`` layouts to a pair of three-part compositions.

    Round *i* is won when every opponent either cannot contribute to it (holding
    fewer than *i* cards of the suit) or contributes something lower.
    """

    mine = sorted(validate_ranks(own_ranks))
    if type(max_leads) is not int or max_leads < 0:
        raise CombinatoricsError("max_leads must be a nonnegative integer")
    outstanding_ranks = _conditioned_outstanding(own_ranks, seen_ranks)
    outstanding = len(outstanding_ranks)
    unseen = _check_conditioning(outstanding, hand_sizes, void)
    filler = unseen - outstanding
    caps = tuple(0 if flag else size for flag, size in zip(void, hand_sizes))

    normaliser = 0
    for counts in _capped_compositions(outstanding, caps):
        remainder = tuple(
            size - count for size, count in zip(hand_sizes, counts)
        )
        normaliser += multinomial(outstanding, counts) * multinomial(filler, remainder)
    if normaliser == 0:
        raise CombinatoricsError(
            "the stated voids and hand sizes admit no placement of the suit"
        )

    rounds = min(max_leads, len(mine))
    totals: list[Fraction] = []
    for index in range(rounds):
        card = mine[index]
        needed = index + 1
        below = sum(1 for rank in outstanding_ranks if rank < card)
        above = outstanding - below
        winning = 0
        for lows in _capped_compositions(below, caps):
            for highs in _capped_compositions(
                above, tuple(cap - low for cap, low in zip(caps, lows))
            ):
                held = tuple(low + high for low, high in zip(lows, highs))
                if any(
                    total >= needed and low < needed
                    for total, low in zip(held, lows)
                ):
                    continue
                remainder = tuple(
                    size - count for size, count in zip(hand_sizes, held)
                )
                winning += (
                    multinomial(below, lows)
                    * multinomial(above, highs)
                    * multinomial(filler, remainder)
                )
        totals.append(Fraction(winning, normaliser))
    return tuple(totals) + (Fraction(0),) * (max_leads - rounds)

def _multinomial(total: int, parts: Sequence[int]) -> int:
    if total < 0 or any(part < 0 for part in parts) or sum(parts) != total:
        return 0
    ways = math.factorial(total)
    for part in parts:
        ways //= math.factorial(part)
    return ways


def _capped_compositions(total: int, caps: tuple[int, int, int]) -> Iterator[tuple]:
    first_cap, second_cap, third_cap = caps
    for first in range(min(total, first_cap) + 1):
        remaining = total - first
        low = max(0, remaining - third_cap)
        for second in range(low, min(remaining, second_cap) + 1):
            third = remaining - second
            if third <= third_cap:
                yield first, second, third


@lru_cache(maxsize=8192)
def ducking_probabilities(
    own_ranks: tuple[int, ...],
    outstanding_ranks: tuple[int, ...],
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool],
    max_leads: int,
    *,
    contested_only: bool = False,
) -> tuple[Fraction, ...]:
    """P(this holding wins round *i*) for i = 1..max_leads, everyone ducking.

    Round *i* is contested by each seat's *i*-th lowest card, and an opponent's
    *i*-th lowest lies below a given card exactly when that opponent holds *i* or
    more cards below it. So a round turns only on how the outstanding ranks split
    into those below the card and those above it, never on which ranks they are -
    which is what replaces an enumeration over rank assignments with a pair of
    three-part compositions.

    With *contested_only*, a round counts only while some opponent can still
    contribute to it. That is the form a liability needs: a round nobody can lead
    is not a trick anybody takes.
    """

    mine = sorted(own_ranks)
    unseen = sum(hand_sizes)
    outstanding = len(outstanding_ranks)
    if outstanding > unseen:
        raise AgentsValidationError("more of the suit is outstanding than is unseen")
    filler = unseen - outstanding
    caps = tuple(0 if flag else size for flag, size in zip(void, hand_sizes))

    normaliser = 0
    for counts in _capped_compositions(outstanding, caps):
        remainder = tuple(size - count for size, count in zip(hand_sizes, counts))
        normaliser += _multinomial(outstanding, counts) * _multinomial(filler, remainder)
    if normaliser == 0:
        raise AgentsValidationError("the public record admits no placement of the suit")

    rounds = min(max_leads, len(mine))
    totals: list[Fraction] = []
    for index in range(rounds):
        card = mine[index]
        needed = index + 1
        below = sum(1 for rank in outstanding_ranks if rank < card)
        above = outstanding - below
        winning = 0
        for lows in _capped_compositions(below, caps):
            highs_caps = tuple(cap - low for cap, low in zip(caps, lows))
            for highs in _capped_compositions(above, highs_caps):
                held = tuple(low + high for low, high in zip(lows, highs))
                if contested_only and max(held) < needed:
                    continue
                if any(
                    total >= needed and low < needed
                    for total, low in zip(held, lows)
                ):
                    continue
                remainder = tuple(
                    size - count for size, count in zip(hand_sizes, held)
                )
                winning += (
                    _multinomial(below, lows)
                    * _multinomial(above, highs)
                    * _multinomial(filler, remainder)
                )
        totals.append(Fraction(winning, normaliser))
    return tuple(totals) + (Fraction(0),) * (max_leads - rounds)


def expected_ducking_tricks(
    own_ranks: tuple[int, ...],
    outstanding_ranks: tuple[int, ...],
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool],
) -> Fraction:
    """Expected tricks this holding takes while the suit is still contested."""

    if not own_ranks:
        return Fraction(0)
    return sum(
        ducking_probabilities(
            own_ranks, outstanding_ranks, hand_sizes, void,
            len(own_ranks), contested_only=True,
        ),
        Fraction(0),
    )


def card_liability(
    rank: int,
    own_ranks: tuple[int, ...],
    outstanding_ranks: tuple[int, ...],
    hand_sizes: tuple[int, int, int],
    void: tuple[bool, bool, bool],
) -> Fraction:
    """Expected tricks that keeping *rank* costs, over discarding it.

    The card discarded joins the outstanding pool only in the sense that it is
    gone from this hand; it cannot win a later trick for anybody, so it is
    dropped from the analysis rather than handed to an opponent.
    """

    if rank not in own_ranks:
        raise AgentsValidationError("the card must be held to be discarded")
    kept = expected_ducking_tricks(own_ranks, outstanding_ranks, hand_sizes, void)
    without = tuple(value for value in own_ranks if value != rank)
    shed = expected_ducking_tricks(without, outstanding_ranks, hand_sizes, void)
    return kept - shed

def _played_cards(observation: PlayerObservation) -> list[tuple[int, Card]]:
    plays: list[tuple[int, Card]] = []
    for trick in observation.completed_tricks:
        for play in trick.plays:
            plays.append((play.seat, play.card))
    for play in observation.current_trick:
        plays.append((play.seat, play.card))
    return plays


def opponent_order(seat: int) -> tuple[int, int, int]:
    """The three other seats, in a fixed clockwise order from *seat*."""

    return ((seat + 1) % 4, (seat + 2) % 4, (seat + 3) % 4)


def public_conditioning(observation: PlayerObservation, suit: Suit) -> tuple:
    """Outstanding ranks, hand sizes and proven voids for *suit*.

    Only played cards and proven voids are used. A card shown at deal start but
    not yet played stays outstanding with an unknown owner, which is the
    documented weakening in this module's docstring.
    """

    seat = observation.observing_seat
    others = opponent_order(seat)
    plays = _played_cards(observation)

    played_in_suit = {card.rank.value for _, card in plays if card.suit is suit}
    held = {card.rank.value for card in observation.own_hand if card.suit is suit}
    outstanding = tuple(
        sorted(
            (
                rank
                for rank in range(2, 15)
                if rank not in played_in_suit and rank not in held
            ),
            reverse=True,
        )
    )

    counts = {other: 0 for other in others}
    for actor, _ in plays:
        if actor in counts:
            counts[actor] += 1
    hand_sizes = tuple(CARDS_PER_HAND - counts[other] for other in others)

    # A seat is void when it failed to follow a suit that was led.
    void_seats: set[int] = set()
    groups = [trick.plays for trick in observation.completed_tricks]
    if observation.current_trick:
        groups.append(tuple(observation.current_trick))
    for group in groups:
        if not group or group[0].card.suit is not suit:
            continue
        for play in group[1:]:
            if play.card.suit is not suit:
                void_seats.add(play.seat)
    void = tuple(other in void_seats for other in others)
    return outstanding, hand_sizes, void


def is_nolo_discard(observation: PlayerObservation) -> bool:
    """True when this policy's rule applies: our NOLO discard, void in the led suit."""

    if (
        observation.phase is not EnginePhase.TRICK_PLAY
        or observation.contract is not Contract.NOLO
        or observation.current_actor != observation.observing_seat
        or not observation.current_trick
    ):
        return False
    led = observation.current_trick[0].card.suit
    return all(card.suit is not led for card in observation.own_hand)


def priced_discard(observation: PlayerObservation) -> Card:
    """The card to discard: the one whose retention costs the most tricks.

    Ties break deterministically - the larger liability, then the shorter suit,
    then the higher rank, then canonical suit order - so the rule carries no
    hidden randomness.
    """

    by_suit: dict[Suit, list[Card]] = {}
    for card in observation.own_hand:
        if not isinstance(card, Card):
            raise AgentsValidationError("a discard requires Card values")
        by_suit.setdefault(card.suit, []).append(card)
    if not by_suit:
        raise AgentsValidationError("a discard requires a non-empty hand")

    best: tuple | None = None
    chosen: Card | None = None
    for suit, cards in by_suit.items():
        outstanding, hand_sizes, void = public_conditioning(observation, suit)
        own = tuple(sorted(card.rank.value for card in cards))
        for card in cards:
            liability = card_liability(
                card.rank.value, own, outstanding, hand_sizes, void
            )
            key = (liability, -len(cards), card.rank.value, -_suit_order(card.suit))
            if best is None or key > best:
                best, chosen = key, card
    if chosen is None:
        raise AgentsValidationError("no discard could be priced")
    return chosen


def _suit_order(suit: Suit) -> int:
    return list(Suit).index(suit)

            priced_discard(observation),

        )

class _ReplayMeter:
    def __init__(self, limit):
        self.limit = limit
        self.calls = 0
        # The existing replay has no transition hook. Rebind only this call's
        # globals, keeping its code/defaults/closure and helper functions intact.
        # Unlike monkeypatching, this cannot alter concurrent callers' engine.
        original = hypothetical_continuation
        namespace = dict(original.__globals__)
        apply = namespace["apply_action"]

        def counted_apply(state, action):
            if self.calls >= self.limit:
                raise _TransitionLimit
            self.calls += 1  # Includes attempted engine calls that reject input.
            return apply(state, action)

        namespace["apply_action"] = counted_apply
        self.replay = FunctionType(original.__code__, namespace, original.__name__,
                                   original.__defaults__, original.__closure__)
        self.replay.__kwdefaults__ = original.__kwdefaults__


def _assignments(model: HardConstraintModel, obs: PlayerObservation) -> Iterator[HiddenHandAssignment]:
    cards = sorted(model.unknown_cards,
                   key=lambda c: (len(model.allowed_seats(c)), card_to_index(c)))
    hands = [set(h) for h in model.known_held]
    capacity = list(model.capacity)
    owners = {}
    index = 0

    def visit(position):
        nonlocal index
        if position == len(cards):
            if any(capacity):
                raise SupportInvariantError("enumeration left unfilled capacity")
            ordered = tuple(obs.own_hand if s == obs.observing_seat else
                            tuple(sorted(hand, key=card_to_index))
                            for s, hand in enumerate(hands))
            yield HiddenHandAssignment(obs.observing_seat, ordered,
                                       MappingProxyType(dict(owners)), index)
            index += 1
            return
        card = cards[position]
        for seat in sorted(model.allowed_seats(card)):
            if capacity[seat]:
                hands[seat].add(card)
                owners[card] = seat
                capacity[seat] -= 1
                yield from visit(position + 1)
                capacity[seat] += 1
                del owners[card]
                hands[seat].remove(card)

    yield from visit(0)


def _validate_public_record(obs):
    """Check public consistency before interpreting replay rejection as a bug."""

    ):

    if obs.forehand != (obs.dealer + 1) % 4 or obs.first_leader != obs.forehand:
        raise ConstraintError("inconsistent first leader/forehand")
    if [a.seat for a in obs.show_actions] != [(obs.forehand + i) % 4 for i in range(4)]:
        raise ConstraintError("showing order contradicts dealer")
    played_owner = {}
    voids = [set() for _ in range(4)]
    groups = tuple(t.plays for t in obs.completed_tricks) + (obs.current_trick,)
    for plays in groups:
        if not plays:

                raise ConstraintError("repeated card or play after a proven void")
            played_owner[play.card] = play.seat
            if play.card.suit is not led:
                voids[play.seat].add(led)
    if len({a.shown_card for a in obs.show_actions}) != 4:
        raise ConstraintError("a card cannot be shown by two seats")
    for action in obs.show_actions:
        owner = played_owner.get(action.shown_card)
        if owner is not None and owner != action.seat:
            raise ConstraintError("shown card was played by a different owner")
    leader = obs.first_leader
    totals = [0, 0]
    for index, trick in enumerate(obs.completed_tricks):
        winner = trick_winner(trick.plays)
        if (trick.trick_index != index or trick.leader != leader or
                trick.plays[0].seat != leader or trick.winner != winner):
            raise ConstraintError("contradictory completed trick")
        totals[winner % 2] += 1
        leader = winner
    if tuple(totals) != obs.team_tricks or len(obs.current_trick) > 3:
        raise ConstraintError("contradictory trick totals/partial trick")
    if any(p.seat != (leader + i) % 4 for i, p in enumerate(obs.current_trick)):

def build_support(observation: PlayerObservation, budget: SupportBudget) -> SupportResult:
    """Return the complete hard support or a typed unresolved result, never a sample.

    Wrong entry-point types are programmer errors. Contradictory evidence returns
    INVALID_INPUT. Rejection/count disagreement within exact enumeration raises
    SupportInvariantError, and unexpected errors propagate unchanged.
    """
    if not isinstance(observation, PlayerObservation):
        raise TypeError("build_support requires PlayerObservation, not a private state/fixture")
    if not isinstance(budget, SupportBudget):
        raise TypeError("build_support requires SupportBudget")
    key = semantic_observation_key(observation)
    counted = None
    attempted = 0
    meter = _ReplayMeter(budget.max_engine_transitions)

    def unresolved(status, reason):
        return SupportResult(status, key, None, counted, attempted, meter.calls, reason)

    if observation.phase is not EnginePhase.TRICK_PLAY or observation.contract is not Contract.NOLO:
        return unresolved(SupportStatus.OUT_OF_SCOPE, "requires resolved NOLO trick play")
    remaining = 13 - len(observation.completed_tricks)
    if not 1 <= remaining <= budget.max_remaining_tricks:
        return unresolved(SupportStatus.OUT_OF_SCOPE, "remaining-trick limit")
    try:
        _validate_public_record(observation)
        model = HardConstraintModel.build(observation)
    except (ConstraintError, AgentsValidationError, RulesError, StateInvariantError) as exc:
        return unresolved(SupportStatus.INVALID_INPUT, str(exc))
    counted = feasible_assignment_count(model)
    if counted == 0:
        return unresolved(SupportStatus.INVALID_INPUT, "no hard-feasible assignments")
    if counted > budget.max_worlds:
        return unresolved(SupportStatus.SUPPORT_BUDGET, "complete count exceeds world limit")
    worlds = []
    seen = set()
    for assignment in _assignments(model, observation):
        attempted += 1
        try:
            continuation = meter.replay(observation, assignment)
        except _TransitionLimit:
            return unresolved(SupportStatus.SEARCH_BUDGET, "engine transition limit during replay")
        except (EngineValidationError, RulesError, ContinuationError) as exc:
            raise SupportInvariantError(
                f"hard-model candidate {assignment.content_key()} rejected after "
                f"{meter.calls} engine calls: {exc}"
            ) from exc
        replayed_key = semantic_observation_key(observation_for(continuation.state, observation.observing_seat))
        if replayed_key != key:
            raise SupportInvariantError("replay changed semantic observation")
        world = ResearchWorld(assignment, continuation)
        if world.content_key in seen:
            raise SupportInvariantError("duplicate assignment")
        seen.add(world.content_key)
        worlds.append(world)
    if len(worlds) != counted:
        raise SupportInvariantError(f"enumerated {len(worlds)} assignments, counted {counted}")
    return SupportResult(SupportStatus.EXACT_COMPLETE, key, counted, counted,
                         attempted, meter.calls, None, tuple(worlds))

"""N04: an information-legal security continuation for one controlled seat.

The controlled seat commits to **one** action per information key. Every other
seat, partner included, is quantified over arbitrary legal actions. That is a
robust guarantee, not the cooperative team value and not an equilibrium: an
unhelpful partner is inside the model on purpose.

Why per-world minimisation is invalid, in one line: choosing separately in each
world lets the plan use information the seat does not have, so its value can
beat anything actually achievable. The recursion below therefore intersects
legal actions across an information class and picks a single action for the
class, while the adversary is free to differentiate.

class _Meter:
    def __init__(self, limit: int, already_used: int, backend=None) -> None:
        self.limit = limit
        self.calls = already_used
        self.nodes = 0
        execution = None if backend is None else backend.execution()
        self.apply = apply_action if execution is None else execution.apply_action
        self.observe = observation_for if execution is None else execution.observation_for
        self.information_key = semantic_observation_key if execution is None else execution.information_key

    def step(self, state, action):
        if self.calls >= self.limit:
            raise _Budget
        self.calls += 1
        return self.apply(state, action)


def _terminal_value(states, team: int) -> int:
    """Worst case inside the class: nature keeps the least favourable world."""

    return max(state.team_tricks[team] for state in states)


def _ranked_actions(agent, observation) -> tuple[PlayCardAction, ...]:
    """The inherited v2 preference order, used only to settle exact ties."""

    ranking = agent.decide(observation, diagnostics=False).scored
    return tuple(
        item.action for item in ranking.ranked() if isinstance(item.action, PlayCardAction)
    )


def _preference_index(order: tuple[PlayCardAction, ...], action: PlayCardAction) -> int:
    try:
        return order.index(action)
    except ValueError:
        return len(order)


def _search(states, *, seat: int, team: int, meter: _Meter, agents) -> tuple[int, dict]:
    """Value of one information class, and the own plan achieving it.

    ``states`` are the hypothetical states of every world still compatible with
    the public history. They necessarily share a current actor and, at an own
    turn, a single information key.
    """

    meter.nodes += 1
    if states[0].phase is not EnginePhase.TRICK_PLAY:
        return _terminal_value(states, team), {}

    actor = states[0].current_actor
    if actor != seat:
        # Every other seat, partner included, may do anything legal. Each public
        # action keeps exactly the worlds in which it was available.
        union: dict = {}
        for state in states:
            for action in meter.observe(state, actor).legal_actions:
                union.setdefault(action, []).append(state)
        worst = -1
        merged: dict = {}
        for action, subset in union.items():
            successors = tuple(meter.step(state, action) for state in subset)
            value, plan = _search(
                successors, seat=seat, team=team, meter=meter, agents=agents
            )
            worst = max(worst, value)
            # Every adversary branch is reachable, so all of their own-plan
            # entries belong in the exported plan.
            for key, chosen in plan.items():
                if merged.setdefault(key, chosen) != chosen:
                    raise SecurityInvariantError(
                        "two reachable branches demand different actions at one key"
                    )
        return worst, merged

    # The controlled seat. Partition defensively: distinct observation classes
    # are distinct decisions, and nature picks the worst of them.
    classes: dict = {}
    for state in states:
        observation = meter.observe(state, seat)
        classes.setdefault(meter.information_key(observation), []).append(
            (state, observation)
        )

    worst = -1
    merged = {}
    for key, members in classes.items():
        common = set(members[0][1].legal_actions)
        for _state, observation in members[1:]:
            common &= set(observation.legal_actions)
        if not common:
            raise SecurityInvariantError("an information class shares no legal action")
        order = _ranked_actions(agents[seat], members[0][1])
        best_value = None
        best_action = None
        best_plan: dict = {}
        # One action for the whole class. Never minimise per world.
        for action in sorted(common, key=lambda a: _preference_index(order, a)):
            successors = tuple(meter.step(state, action) for state, _obs in members)
            value, plan = _search(
                successors, seat=seat, team=team, meter=meter, agents=agents
            )
            if best_value is None or value < best_value:
                best_value, best_action, best_plan = value, action, plan
        for key_inner, chosen in best_plan.items():
            if merged.setdefault(key_inner, chosen) != chosen:
                raise SecurityInvariantError(
                    "two reachable branches demand different actions at one key"
                )
        # A forced action still belongs in the plan only if the verifier will
        # look for it; it consults every own node, so record every one.
        merged[key] = best_action
        worst = max(worst, best_value)
    return worst, merged


def _search_cached(states, *, seat, team, meter, agents, cache, root_key):
    """The memoised equivalent of :func:`_search`, in two passes.

    Values first, which is where repetition can be exploited, then a plan

def solve_security(
    observation: PlayerObservation,
    budget: SupportBudget,
    *,
    verification_budget: SupportBudget | None = None,
    cache=None,
    backend=None,
) -> SecurityResult:
    """Return every root action's security value, a plan, and its verification.

    The plan is checked by N03's independent verifier rather than trusted. The
    verification runs on its own budget and is reported separately, so it is
    never presented as free computation.

    Passing a ``SecurityCache`` selects the N06 memoised path. It must agree
    with the plain path on every value, label and action; the uncached search
    below stays the independent reference and remains the default.
    """

    if not isinstance(observation, PlayerObservation):
        raise TypeError("solve_security requires PlayerObservation, not a private state")
    if not isinstance(budget, SupportBudget):
        raise TypeError("solve_security requires SupportBudget")
    if backend is not None and cache is not None:
        raise ValueError("N06 cache is supported only by the reference backend")
    backend_id = "reference_engine" if backend is None else backend.identity

    key = semantic_observation_key(observation)

    def unresolved(status: SecurityStatus, reason: str, used: int, nodes: int = 0):
        return SecurityResult(
            status, key, (), None, (), None, None, None, used, 0, nodes, reason,
            execution_backend=backend_id,
        )

    if observation.current_actor != observation.observing_seat:
        return unresolved(
            SecurityStatus.INVALID_INPUT, "root observer must be the acting seat", 0
        )

    support = (build_support if backend is None else backend.build_support)(observation, budget)
    if support.status is not SupportStatus.EXACT_COMPLETE:
        return unresolved(
            SecurityStatus(support.status.value),
            support.reason,
            support.engine_transitions,
        )

    seat = observation.observing_seat
    team = seat % 2
    agents = void_seeking_agents()
    meter = _Meter(budget.max_engine_transitions, support.engine_transitions, backend)
    roots = tuple(
        action for action in observation.legal_actions if isinstance(action, PlayCardAction)
    )
    order = _ranked_actions(agents[seat], observation)

    values: list[RootValue] = []
    plans: dict = {}
    try:
        for action in roots:
            successors = tuple(
                meter.step(world.continuation.state, action)
                for world in support.research_worlds
            )
            if cache is None:
                value, plan = _search(
                    successors, seat=seat, team=team, meter=meter, agents=agents
                )
            else:
                value, plan = _search_cached(
                    successors, seat=seat, team=team, meter=meter,
                    agents=agents, cache=cache, root_key=key,
                )
            values.append(RootValue(action, value))
            plans[action] = plan
    except _Budget:
        return unresolved(
            SecurityStatus.SEARCH_BUDGET,
            "engine transition limit during security search",
            meter.calls,
            meter.nodes,
        )

    best = min(v.security_value for v in values)
    selected = min(
        (v for v in values if v.security_value == best),
        key=lambda v: _preference_index(order, v.action),
    ).action

    plan = tuple(
        PlanEntry(information_key=k, action=a) for k, a in sorted(plans[selected].items())
    )
    certificate = Certificate(
        source_information_key=key,
        root_seat=seat,
        root_action=selected,
        objective=OBJECTIVE,
        control_model=ControlModel.SINGLE_SEAT,
        complete_support_digest=(complete_support_digest if backend is None else backend.support_digest)(support),
        plan=plan,
        claimed_terminal_bound=best,
        provenance=CompletenessProvenance(
            support_status=support.status,
            world_count=support.world_count,
            generation_budget=budget,
            generation_engine_transitions=support.engine_transitions,
        ),
    )
    checking = verification_budget or budget
    verification = verify_certificate(observation, certificate, checking, backend=backend)

    return SecurityResult(
        status=SecurityStatus.EXACT_COMPLETE,
        information_key=key,
        root_values=tuple(values),
        selected_action=selected,
        plan=plan,
        certificate=certificate,
        verification=verification,
        world_count=support.world_count,
        search_engine_transitions=meter.calls,
        verification_engine_transitions=verification.engine_transitions,
        nodes_visited=meter.nodes,
        reason=None,
        execution_backend=backend_id,
    )

"""The hard public constraint model: every exact exclusion, and nothing else.

Phase 4's :class:`tuppi_agents.PublicInference` answers *"is this excluded by
public information?"* from five facts. This module is the Phase 5B layer that
sits directly on top of it and turns those facts into the explicit combinatorial
object a counter, a sampler, and a solver can consume:

* which seats each unknown card may still belong to;
* exactly how many unknown cards each seat must still receive.

It adds one class of exclusion that Phase 4 did not model, and that omission was
not cosmetic: **a show action constrains the shower's whole initial hand.**

``project_target_v1`` derives ``verbal_required`` from the complete 13-card hand
through a mandatory three-level hierarchy, so the public
``(seat, shown_card, declared_contract, verbal_required)`` tuple is a hard
statement about cards the shower does *not* hold:

============================================  ===========================================
Public show action                             Exact deduction about the initial hand
============================================  ===========================================
``verbal_required`` is false                   Priority 1 applied. The shown card is itself
                                               an eligible card of the contract's normal
                                               colour, so nothing new is excluded.
``verbal_required`` and the shown rank is       Priority 2 applied. The hand holds **no**
eligible (3-9)                                 eligible card of the contract's normal
                                               colour: 14 cards are excluded.
``verbal_required`` and the shown rank is       Priority 3 applied. The hand holds **no**
ineligible                                     eligible card of *either* colour: all 28
                                               cards of ranks 3-9 are excluded.
============================================  ===========================================

Priority 3 cannot show an eligible card, because under priority 3 the hand
contains none; that is why the shown card's rank separates the two fallback
levels with certainty rather than by guess.

The deduction is about the *initial* hand, but every card that seat has played is
already public, so the only live consequence is an exclusion over the seat's
remaining unknown cards. A legal state can never show an excluded card as
already played by that seat, and :func:`derive_show_exclusions` treats that
combination as a contradiction rather than silently dropping it.

There is no probability anywhere in this module. An unknown card's allowed-seat
set is the full set the facts permit, with no ordering and no weight among them.
Counting and sampling live in :mod:`tuppi_exact.counting` and
:mod:`tuppi_exact.sampling`; a *count* over that set is a combinatorial fact
about the constraint system, never a claim about what an opponent would do.
"""

#: The full set of cards whose rank is show-eligible under ``project_target_v1``.
ELIGIBLE_SHOW_CARDS: frozenset[Card] = frozenset(
    card for card in FULL_DECK if is_show_rank_eligible(card.rank)
)

#: Eligible show cards partitioned by colour, precomputed once.
ELIGIBLE_SHOW_CARDS_BY_COLOR: Mapping[Color, frozenset[Card]] = MappingProxyType(
    {
        color: frozenset(
            card for card in ELIGIBLE_SHOW_CARDS if card_color(card) is color
        )
        for color in Color
    }
)
_SUIT_ORDER = {suit: index for index, suit in enumerate(Suit)}


def show_exclusion_for(action: ShowAction) -> frozenset[Card]:
    """Return the cards one public show action proves are absent from the hand.

    The result is a statement about the shower's complete initial hand and is a
    direct consequence of the mandatory showing hierarchy, not an inference
    about the shower's intentions.
    """

    if not isinstance(action, ShowAction):
        raise ConstraintError("show_exclusion_for requires a rules ShowAction")
    if not action.verbal_required:
        return frozenset()
    if is_show_rank_eligible(action.shown_card.rank):
        # Priority 2: no eligible card of the contract's normal colour exists.
        return ELIGIBLE_SHOW_CARDS_BY_COLOR[contract_color(action.declared_contract)]
    # Priority 3: no eligible card of either colour exists.
    return ELIGIBLE_SHOW_CARDS


def derive_show_exclusions(
    observation: PlayerObservation,
) -> tuple[frozenset[Card], ...]:
    """Return the per-seat show-derived exclusion sets for one observation.

    The observing seat's own entry is always empty: its hand is known outright,
    so an exclusion over it would be redundant, and applying one would risk
    contradicting the very hand the engine handed over.
    """

    if not isinstance(observation, PlayerObservation):
        raise ConstraintError("derive_show_exclusions requires a PlayerObservation")
    exclusions: list[set[Card]] = [set(), set(), set(), set()]
    played_by_seat: list[set[Card]] = [set(), set(), set(), set()]
    for trick in observation.completed_tricks:
        for play in trick.plays:
            played_by_seat[play.seat].add(play.card)
    for play in observation.current_trick:
        played_by_seat[play.seat].add(play.card)

    for action in observation.show_actions:
        if action.seat == observation.observing_seat:
            continue
        excluded = show_exclusion_for(action)
        contradiction = excluded & played_by_seat[action.seat]
        if contradiction:
            raise ConstraintError(
                "a show-derived exclusion contradicts a card that seat already "
                "played; the public record is not a legal project_target_v1 state"
            )
        exclusions[action.seat] |= excluded
    return tuple(frozenset(item) for item in exclusions)


@dataclass(frozen=True, slots=True)
class HardConstraintModel:
    """An exact, immutable description of every feasible hidden-hand assignment.

    The model is a degree-constrained bipartite structure: unknown cards on one
    side, seats with exact residual capacities on the other, and an edge
    wherever public information permits. Nothing in it is approximate, and no
    field carries a weight, a prior, or a likelihood.
    """

    observing_seat: int
    unknown_cards: tuple[Card, ...]
    allowed_seat_map: Mapping[Card, frozenset[int]]
    capacity: tuple[int, ...]
    known_held: tuple[frozenset[Card], ...]
    played_cards: frozenset[Card]
    void_suits: tuple[frozenset[Suit], ...]
    show_exclusions: tuple[frozenset[Card], ...]
    cards_remaining: tuple[int, ...]
    refinement_passes: int
    schema_version: str = field(
        default=CONSTRAINT_MODEL_SCHEMA_VERSION, init=False
    )

    def __post_init__(self) -> None:
        if type(self.observing_seat) is not int or self.observing_seat not in range(4):
            raise ConstraintError("observing seat must be an integer in 0..3")
        for label, value in (
            ("capacity", self.capacity),
            ("known_held", self.known_held),
            ("void_suits", self.void_suits),
            ("show_exclusions", self.show_exclusions),
            ("cards_remaining", self.cards_remaining),
        ):
            if type(value) is not tuple or len(value) != 4:
                raise ConstraintError(f"{label} must be a four-seat tuple")
        if any(type(value) is not int or value < 0 for value in self.capacity):
            raise ConstraintError("capacities must be nonnegative integers")
        if self.capacity[self.observing_seat] != 0:
            raise ConstraintError("the observing seat has no unknown capacity")
        if sum(self.capacity) != len(self.unknown_cards):
            raise ConstraintError(
                "total residual capacity disagrees with the unknown card count"
            )
        if len(set(self.unknown_cards)) != len(self.unknown_cards):
            raise ConstraintError("unknown cards must be distinct")
        if set(self.allowed_seat_map) != set(self.unknown_cards):
            raise ConstraintError("allowed-seat map must cover exactly the unknowns")
        for card, seats in self.allowed_seat_map.items():
            if not isinstance(seats, frozenset) or not seats:
                raise ConstraintError(
                    "every unknown card needs a nonempty frozen allowed-seat set"
                )
            if any(seat not in range(4) for seat in seats):
                raise ConstraintError("allowed seats must be in 0..3")
            if any(self.capacity[seat] == 0 for seat in seats):
                raise ConstraintError(
                    "an allowed seat with zero residual capacity was not propagated"
                )
            if card in self.played_cards:
                raise ConstraintError("a played card cannot be unknown")
        for seat in range(4):
            if self.known_held[seat] & self.played_cards:
                raise ConstraintError("a played card cannot be publicly held")
            if len(self.known_held[seat]) + self.capacity[seat] != (
                self.cards_remaining[seat]
            ):
                raise ConstraintError(
                    "known cards plus residual capacity must equal the hand size"
                )
            if self.known_held[seat] & self.show_exclusions[seat]:
                raise ConstraintError(
                    "a show-derived exclusion contradicts a publicly held card"
                )
            if any(
                card.suit in self.void_suits[seat] for card in self.known_held[seat]
            ):
                raise ConstraintError("a proven void contradicts a publicly held card")

    # ---------------------------------------------------------------- builders

    @classmethod
    def build(cls, observation: PlayerObservation) -> "HardConstraintModel":
        """Derive the model from one player observation.

        The Phase 4 inference object supplies the five original facts. This
        method intersects them with the show-derived exclusions and re-runs the
        same two sound deductions - singleton and pigeonhole - plus the seat
        subset (Hall) propagation that exact capacities make available, to a
        fixed point.
        """

        inference = PublicInference.build(observation)
        exclusions = derive_show_exclusions(observation)
        return cls._from_facts(inference=inference, show_exclusions=exclusions)

    @classmethod
    def from_inference(
        cls,
        inference: PublicInference,
        show_exclusions: tuple[frozenset[Card], ...],
    ) -> "HardConstraintModel":
        """Build directly from an existing inference plus explicit exclusions."""

        if not isinstance(inference, PublicInference):
            raise ConstraintError("from_inference requires a PublicInference")
        if type(show_exclusions) is not tuple or len(show_exclusions) != 4:
            raise ConstraintError("show_exclusions must be a four-seat tuple")
        return cls._from_facts(inference=inference, show_exclusions=show_exclusions)

    @classmethod
    def _from_facts(
        cls,
        *,
        inference: PublicInference,
        show_exclusions: tuple[frozenset[Card], ...],
    ) -> "HardConstraintModel":
        seat = inference.observing_seat
        known: list[set[Card]] = [set(item) for item in inference.known_held]
        capacity = list(inference.unknown_capacity)
        allowed: dict[Card, frozenset[int]] = {}
        for card in inference.unaccounted_cards:
            seats = frozenset(
                candidate
                for candidate in inference.possible_owners(card)
                if card not in show_exclusions[candidate]
            )
            if not seats:
                raise ConstraintError(
                    "hard public constraints exclude every holder of an unknown card"
                )
            allowed[card] = seats

        passes = _refine(allowed=allowed, capacity=capacity, known=known)

        unknown = tuple(sorted(allowed, key=_card_key))
        return cls(
            observing_seat=seat,
            unknown_cards=unknown,
            allowed_seat_map=MappingProxyType(dict(allowed)),
            capacity=tuple(capacity),
            known_held=tuple(frozenset(item) for item in known),
            played_cards=inference.played_cards,
            void_suits=inference.void_suits,
            show_exclusions=tuple(show_exclusions),
            cards_remaining=inference.cards_remaining,
            refinement_passes=passes,
        )

    # ----------------------------------------------------------------- queries

    @property
    def unknown_seats(self) -> tuple[int, ...]:
        """Seats that must still receive at least one unknown card."""

        return tuple(seat for seat in range(4) if self.capacity[seat] > 0)

    def allowed_seats(self, card: Card) -> frozenset[int]:
        """Return every seat the hard facts still permit to hold ``card``.

        Empty for a played card and for a card whose owner is already proven -
        in the latter case :meth:`proven_owner` answers instead.
        """

        if not isinstance(card, Card):
            raise ConstraintError("allowed_seats requires a Card")
        return self.allowed_seat_map.get(card, frozenset())

    def proven_owner(self, card: Card) -> int | None:
        """Return the seat publicly proven to hold ``card``, or ``None``."""

        if not isinstance(card, Card):
            raise ConstraintError("proven_owner requires a Card")
        for seat in range(4):
            if card in self.known_held[seat]:
                return seat
        return None

    def equivalence_classes(self) -> tuple[tuple[frozenset[int], tuple[Card, ...]], ...]:
        """Group unknown cards by their allowed-seat set.

        Two unknown cards with the same allowed-seat set are interchangeable for
        every feasibility, counting, and uniform-sampling question. Collapsing
        them is what makes the exact counter cheap, and it is exact rather than
        an approximation: the constraint system genuinely cannot distinguish
        them.
        """

        grouped: dict[frozenset[int], list[Card]] = {}
        for card in self.unknown_cards:
            grouped.setdefault(self.allowed_seat_map[card], []).append(card)
        return tuple(
            (seats, tuple(sorted(cards, key=_card_key)))
            for seats, cards in sorted(
                grouped.items(), key=lambda item: tuple(sorted(item[0]))
            )
        )

    def hall_deficiency(self) -> tuple[tuple[tuple[int, ...], int, int], ...]:
        """Return every seat subset whose capacity cannot absorb its captive cards.

        A card is *captive* to a seat subset when no seat outside that subset may
        hold it. With at most three unknown seats, enumerating all subsets is a
        complete feasibility test, not a heuristic screen.
        """

        seats = self.unknown_seats
        failures: list[tuple[tuple[int, ...], int, int]] = []
        for size in range(1, len(seats) + 1):
            for subset in combinations(seats, size):
                chosen = frozenset(subset)
                captive = sum(
                    1
                    for card in self.unknown_cards
                    if self.allowed_seat_map[card] <= chosen
                )
                available = sum(self.capacity[item] for item in subset)
                if captive > available:
                    failures.append((subset, captive, available))
        return tuple(failures)

    def is_structurally_feasible(self) -> bool:
        """True when no seat subset is over-subscribed.

        This is the complete Gale-Hoffman condition for the transportation
        problem the model describes, so for ``project_target_v1`` - where at most
        three seats are ever unknown - it is exact. The authoritative oracle is
        still :func:`tuppi_exact.counting.feasible_assignment_count`, which
        returns zero for an infeasible model.
        """

        return not self.hall_deficiency()

    def identity_data(self) -> dict[str, object]:
        """Canonical content for digesting a model without revealing a hand."""

        from tuppi_rules import card_to_code

        return {
            "allowed_seats": {
                card_to_code(card): sorted(self.allowed_seat_map[card])
                for card in self.unknown_cards
            },
            "capacity": list(self.capacity),
            "observing_seat": self.observing_seat,
            "schema_version": CONSTRAINT_MODEL_SCHEMA_VERSION,
            "unknown_cards": [card_to_code(card) for card in self.unknown_cards],
        }


def _refine(
    *,
    allowed: dict[Card, frozenset[int]],
    capacity: list[int],
    known: list[set[Card]],
) -> int:
    """Run singleton, zero-capacity, tight-seat, and Hall propagation to a fixed point.

    Each deduction is exact. The loop is bounded by the number of unknown cards
    plus a small constant because every pass that changes anything either proves
    one card's owner or removes at least one edge, and both are monotone.
    """

    ordered_cards = tuple(sorted(allowed, key=_card_key))
    limit = len(allowed) * 4 + 8
    for completed in range(limit):
        changed = False

        for card in ordered_cards:
            if card not in allowed:
                continue
            seats = allowed[card]
            if len(seats) == 1:
                (owner,) = tuple(seats)
                known[owner].add(card)
                capacity[owner] -= 1
                if capacity[owner] < 0:
                    raise ConstraintError(
                        "propagation drove a seat's residual capacity negative"
                    )
                del allowed[card]
                changed = True

        for seat in range(4):
            if capacity[seat] != 0:
                continue
            for card in ordered_cards:
                if card not in allowed:
                    continue
                if seat in allowed[card]:
                    reduced = allowed[card] - {seat}
                    if not reduced:
                        raise ConstraintError(
                            "propagation excluded every holder of an unknown card"
                        )
                    allowed[card] = reduced
                    changed = True

        open_seats = tuple(seat for seat in range(4) if capacity[seat] > 0)
        for seat in open_seats:
            candidates = tuple(
                card
                for card in ordered_cards
                if card in allowed and seat in allowed[card]
            )
            if len(candidates) < capacity[seat]:
                raise ConstraintError(
                    "a seat cannot be filled from the cards it may still hold"
                )
            if len(candidates) == capacity[seat]:
                for card in candidates:
                    if allowed[card] != frozenset((seat,)):
                        allowed[card] = frozenset((seat,))
                        changed = True

        for size in range(1, len(open_seats)):
            for subset in combinations(open_seats, size):
                chosen = frozenset(subset)
                captive = tuple(
                    card
                    for card in ordered_cards
                    if card in allowed and allowed[card] <= chosen
                )
                available = sum(capacity[item] for item in subset)
                if len(captive) > available:
                    raise ConstraintError(
                        "a seat subset cannot absorb the cards captive to it"
                    )
                if len(captive) == available:
                    for card in ordered_cards:
                        if card not in allowed:
                            continue
                        if allowed[card] <= chosen:
                            continue
                        reduced = allowed[card] - chosen
                        if not reduced:
                            raise ConstraintError(
                                "Hall propagation excluded every holder of a card"
                            )
                        if reduced != allowed[card]:
                            allowed[card] = reduced
                            changed = True

        if not changed:
            return completed + 1
    raise ConstraintError("hard constraint propagation did not reach a fixed point")


def _card_key(card: Card) -> tuple[int, int]:
    return (_SUIT_ORDER[card.suit], card.rank.value)


def own_show_exclusion_is_consistent(
    observation: PlayerObservation,
) -> bool:
    """True when the observer's own show action agrees with its own hand.

    This is a self-check rather than an inference: the observer knows its hand,
    so a disagreement would mean the engine handed over an inconsistent
    observation, not that a deduction is unsound.
    """

    if not isinstance(observation, PlayerObservation):
        raise ConstraintError("own_show_exclusion_is_consistent needs an observation")
    own_initial = set(observation.own_hand)
    for trick in observation.completed_tricks:
        for play in trick.plays:
            if play.seat == observation.observing_seat:
                own_initial.add(play.card)
    for play in observation.current_trick:
        if play.seat == observation.observing_seat:
            own_initial.add(play.card)
    if len(own_initial) != CARDS_PER_HAND:
        return False
    for action in observation.show_actions:
        if action.seat != observation.observing_seat:
            continue
        if show_exclusion_for(action) & own_initial:
            return False
    return True

"""Hypothetical continuation states: a sampled world made playable.

A sampled hidden-hand assignment is only useful if something can be *played* on
it. This module turns one assignment plus one observation into a complete
:class:`tuppi_engine.DealState` that the clean engine accepts, so a solver or a
rollout runs against the real rules rather than against a private re-implementation.

The construction deliberately takes the long route. It rebuilds the four initial
13-card hands - the observer's real hand, each opponent's sampled cards, plus the
cards each seat has publicly played - and then **replays the entire public action
history through the ordinary engine transitions**. Nothing is stitched together
by hand and no invariant is asserted locally, so every guarantee comes from the
engine that owns it:

* an assignment that violates a proven void makes a replayed follow-suit
  violation, and the engine rejects it;
* an assignment that contradicts a public show action makes ``verbal_required``
  disagree with the hand, and the engine rejects it;
* an assignment that misplaces even one card fails the state's own hand-history
  reconciliation.

That is the design point worth keeping: the engine is the referee for the
sampler's soundness, so an unsound constraint model cannot silently produce a
plausible-looking hypothetical.

**Hypothetical states are labelled, always.** Every continuation carries a
distinct ``match_ref`` prefix, an explicit ``is_hypothetical`` flag, and the
identity of the observation it was derived from. A hypothetical deal must never
enter a result record, a bank, or a report as if it had been dealt. Its
``deal_id`` genuinely differs from the real deal's, because the hands differ -
which is the property that makes the mistake detectable rather than invisible.
"""

from dataclasses import dataclass, field

def reconstruct_initial_hands(
    observation: PlayerObservation, assignment: HiddenHandAssignment
) -> tuple[tuple[Card, ...], ...]:
    """Return the four initial 13-card hands implied by the assignment.

    Each seat's initial hand is exactly what it still holds plus what it has
    already played, both of which are known: the first from the assignment, the
    second from the public record.

    Order matters, for one specific reason. The engine derives legal-action order
    from the order of the cards in a hand, and Phase 4's tie-break is the lowest
    legal-action index, so hand order can change a policy's choice.

    * The **observing seat's** current order is known, so it is preserved
      exactly: its own cards come first in the observed order and its played
      cards are appended, and removing the played cards during replay leaves the
      observed order intact.
    * Every **other seat's** original order is not public and is not even
      meaningful to the observer, so it is canonicalized by deck index. A
      hypothetical therefore *cannot* reproduce the real deal's ``deal_id``, which
      is a feature: it makes a hypothetical structurally impossible to confuse
      with a deal that was actually dealt.
    """

    if not isinstance(observation, PlayerObservation):
        raise ContinuationError("reconstruct_initial_hands requires an observation")
    if not isinstance(assignment, HiddenHandAssignment):
        raise ContinuationError("reconstruct_initial_hands requires an assignment")
    if assignment.observing_seat != observation.observing_seat:
        raise ContinuationError("the assignment belongs to a different seat")

    seat = observation.observing_seat
    played: list[list[Card]] = [[], [], [], []]
    for trick in observation.completed_tricks:
        for play in trick.plays:
            played[play.seat].append(play.card)
    for play in observation.current_trick:
        played[play.seat].append(play.card)

    hands: list[tuple[Card, ...]] = []
    for index in range(4):
        held = tuple(assignment.hands[index])
        overlap = set(held) & set(played[index])
        if overlap:
            raise ContinuationError("a played card was also assigned as held")
        if index == seat:
            ordered = tuple(observation.own_hand) + tuple(played[index])
            if set(ordered) != set(held) | set(played[index]):
                raise ContinuationError(
                    "the assignment altered the observing seat's own hand"
                )
        else:
            ordered = tuple(
                sorted(set(held) | set(played[index]), key=card_to_index)
            )
        if len(ordered) != 13 or len(set(ordered)) != 13:
            raise ContinuationError(
                f"reconstructed hand for seat {index} has {len(set(ordered))} "
                "distinct cards, not 13"
            )
        hands.append(ordered)
    return tuple(hands)


def hypothetical_continuation(
    observation: PlayerObservation,
    assignment: HiddenHandAssignment,
    *,
    match_ref: str | None = None,
) -> HypotheticalContinuation:
    """Build the engine-validated continuation for one sampled assignment.

    The public action history is replayed through :func:`apply_action`, so the
    resulting state is reachable under ``project_target_v1`` by construction.
    """

    if not isinstance(observation, PlayerObservation):
        raise ContinuationError("hypothetical_continuation requires an observation")
    hands = reconstruct_initial_hands(observation, assignment)
    reference = match_ref or _default_match_ref(observation, assignment)
    if not reference.startswith(HYPOTHETICAL_MATCH_REF_PREFIX):
        raise ContinuationError(
            "an explicit continuation match_ref must keep the hypothetical prefix"
        )

    state = create_deal_state(
        ExplicitDeal(dealer=observation.dealer, hands=hands),
        starting_score=observation.match_score,
        match_ref=reference,
        deal_ref=f"{reference}/deal:0",
        deal_number=0,
        event_start_index=0,
    )
    for action in observation.show_actions:
        state = apply_action(
            state,
            EngineShowAction(
                seat=action.seat,
                shown_card=action.shown_card,
                declared_contract=action.declared_contract,
            ),
        )
    for trick in observation.completed_tricks:
        for play in trick.plays:
            state = apply_action(state, PlayCardAction(seat=play.seat, card=play.card))
    for play in observation.current_trick:
        state = apply_action(state, PlayCardAction(seat=play.seat, card=play.card))

    _require_same_decision_point(observation, state)
    return HypotheticalContinuation(
        state=state,
        observing_seat=observation.observing_seat,
        source_match_ref=observation.match_ref,
        source_deal_ref=observation.deal_ref,
        sample_index=assignment.sample_index,
    )


def continuations_for(
    observation: PlayerObservation,
    assignments: tuple[HiddenHandAssignment, ...],
) -> tuple[HypotheticalContinuation, ...]:
    """Build one continuation per assignment, in the order they were drawn."""

    if type(assignments) is not tuple or not assignments:
        raise ContinuationError("continuations_for requires a nonempty tuple")
    return tuple(
        hypothetical_continuation(observation, assignment)
        for assignment in assignments
    )


def _default_match_ref(
    observation: PlayerObservation, assignment: HiddenHandAssignment
) -> str:
    from tuppi_eval.serialization import content_digest

    digest = content_digest(
        {
            "sample_index": assignment.sample_index,
            "schema_version": CONTINUATION_SCHEMA_VERSION,
            "source_deal_ref": observation.deal_ref,
            "source_match_ref": observation.match_ref,
            "tricks_completed": len(observation.completed_tricks),
        }
    )
    return f"{HYPOTHETICAL_MATCH_REF_PREFIX}:{digest[:32]}"


def _require_same_decision_point(
    observation: PlayerObservation, state: DealState
) -> None:
    """Assert that the replay landed on the observation's own decision point."""

    if state.phase is not observation.phase:
        raise ContinuationError(
            f"replay reached {state.phase.value}, not {observation.phase.value}"
        )
    if state.current_actor != observation.current_actor:
        raise ContinuationError("replay reached a different acting seat")
    if state.phase not in (EnginePhase.SHOWING, EnginePhase.TRICK_PLAY):
        raise ContinuationError("a continuation must stop at an open decision")
    if len(state.completed_tricks) != len(observation.completed_tricks):
        raise ContinuationError("replay produced a different completed-trick count")
    if state.current_trick != observation.current_trick:
        raise ContinuationError("replay produced a different partial trick")
    if state.team_tricks != observation.team_tricks:
        raise ContinuationError("replay produced different team trick totals")
    if tuple(state.hands[observation.observing_seat]) != tuple(observation.own_hand):
        raise ContinuationError(
            "replay altered the observing seat's own hand or its order"
        )


=== E029 | ENGINE | src/tuppi_search/fast_trick.py | complete selected dependency ===
"""Validated search-local trick-play state for Phase 5D-C.

The authoritative engine remains the reference implementation.  This module
only represents the already-resolved trick-play suffix of one hypothetical
deal.  It omits engine event logs and repeated whole-state validation, which
are not consumed by the frozen rollout policies.

There are two ways in, and they lose different amounts.
:meth:`FastTrickState.from_observation_assignment` builds from a formal
observation plus one Phase 5B assignment, so it must undo the sampler's hand
canonicalization for the observing seat.  :meth:`FastTrickState.from_deal_state`
builds from an authoritative state the caller already holds — the rollout's
sampled world — and is an exact projection for all four seats.

The representation is immutable and owns no mutable card containers.  Its
transition and observation functions are differential-tested against
``tuppi_engine`` before use in the registered search agent.
"""

from __future__ import annotations

from dataclasses import dataclass, replace

from tuppi_engine import EnginePhase, PlayCardAction, PlayerObservation
from tuppi_engine.state import CompletedTrick
from tuppi_exact.sampling import HiddenHandAssignment
from tuppi_rules import (
    Contract,
    DealScore,
    MatchScore,
    PlayedCard,
    Team,
    is_match_terminal,
    legal_cards,
    match_winner,
    rotate_dealer,
    score_nolo,
    score_rami_for_declarer,
    team_for_seat,
    transition_rising_score,
    trick_winner,
    validate_play,
)


class FastTrickError(Exception):
    """A search-local transition or projection failed its strict contract."""


@dataclass(frozen=True, slots=True)
class FastTrickState:
    """Minimal immutable state for a resolved hypothetical trick-play suffix."""

    match_ref: str
    deal_ref: str
    dealer: int
    forehand: int
    starting_score: MatchScore
    hands: tuple[tuple, ...]
    current_actor: int | None
    show_actions: tuple
    publicly_shown_cards: tuple[tuple[int, object], ...]
    contract: Contract
    declarer: int | None
    first_leader: int
    current_trick: tuple[PlayedCard, ...]
    completed_tricks: tuple[CompletedTrick, ...]
    team_tricks: tuple[int, int]
    phase: EnginePhase = EnginePhase.TRICK_PLAY
    deal_score: DealScore | None = None
    ending_score: MatchScore | None = None
    next_dealer: int | None = None

    @classmethod
    def from_observation_assignment(
        cls, observation: PlayerObservation, assignment: HiddenHandAssignment
    ) -> "FastTrickState":
        if not isinstance(observation, PlayerObservation):
            raise FastTrickError("fast state requires a PlayerObservation")
        if not isinstance(assignment, HiddenHandAssignment):
            raise FastTrickError("fast state requires a hidden-hand assignment")
        if observation.phase is not EnginePhase.TRICK_PLAY:
            raise FastTrickError("fast state requires a trick-play observation")
        if assignment.observing_seat != observation.observing_seat:
            raise FastTrickError("assignment and observation seats disagree")
        if set(assignment.hands[observation.observing_seat]) != set(
            observation.own_hand
        ):
            raise FastTrickError("assignment changed the observing player's hand")
        if observation.contract is None or observation.first_leader is None:
            raise FastTrickError("trick-play observation lacks resolved contract fields")
        hands = [tuple(hand) for hand in assignment.hands]
        # The sampler canonicalizes every assignment hand.  The observer's
        # current order is public and is behaviorally relevant to Phase 4's
        # legal-action tie-break, so restore that one order exactly.
        hands[observation.observing_seat] = tuple(observation.own_hand)
        return cls(
            match_ref=observation.match_ref,
            deal_ref=observation.deal_ref,
            dealer=observation.dealer,
            forehand=observation.forehand,
            starting_score=observation.match_score,
            hands=tuple(hands),
            current_actor=observation.current_actor,
            show_actions=observation.show_actions,
            publicly_shown_cards=observation.publicly_shown_cards,
            contract=observation.contract,
            declarer=observation.declarer,
            first_leader=observation.first_leader,
            current_trick=observation.current_trick,
            completed_tricks=observation.completed_tricks,
            team_tricks=observation.team_tricks,
        )

    @classmethod
    def from_deal_state(cls, state) -> "FastTrickState":
        """Project an authoritative trick-play state onto the search-local one.

        The rollout already holds a full :class:`~tuppi_engine.DealState` — the
        sampled world with the candidate action forced — so it needs no
        observation and no assignment to hand over. Every field below is copied
        or derived, never inferred: ``publicly_shown_cards`` is the same
        derivation ``observation_for`` performs, and the four hands keep the
        engine's own order, which the legal-action tie-break is sensitive to.

        Unlike :meth:`from_observation_assignment` this loses nothing. There is
        no hidden-hand sampling step here and so no canonicalization to undo:
        the projection is exact for all four seats, not just the observer's.
        """

        if state.phase is not EnginePhase.TRICK_PLAY:
            raise FastTrickError(
                f"fast state requires a trick-play deal state, not {state.phase.value}"
            )
        if state.contract is None or state.first_leader is None:
            raise FastTrickError("trick-play state lacks resolved contract fields")
        return cls(
            match_ref=state.match_ref,
            deal_ref=state.deal_ref,
            dealer=state.dealer,
            forehand=state.forehand,
            starting_score=state.starting_score,
            hands=tuple(tuple(hand) for hand in state.hands),
            current_actor=state.current_actor,
            show_actions=tuple(state.show_actions),
            publicly_shown_cards=tuple(
                (action.seat, action.shown_card) for action in state.show_actions
            ),
            contract=state.contract,
            declarer=state.declarer,
            first_leader=state.first_leader,
            current_trick=tuple(state.current_trick),
            completed_tricks=tuple(state.completed_tricks),
            team_tricks=state.team_tricks,
        )

    @property
    def complete(self) -> bool:
        return self.phase in (EnginePhase.DEAL_COMPLETE, EnginePhase.MATCH_COMPLETE)

    @property
    def terminal(self) -> bool:
        return self.phase is EnginePhase.MATCH_COMPLETE

    def observation_for(self, observing_seat: int) -> PlayerObservation:
        """Project exactly the formal observation consumed by Phase 4 agents."""

        if type(observing_seat) is not int or observing_seat not in range(4):
            raise FastTrickError("observation seat must be an integer in 0..3")
        legal_actions = self._legal_actions_for(observing_seat)
        current_score = self.ending_score or self.starting_score
        return PlayerObservation(
            match_ref=self.match_ref,
            deal_ref=self.deal_ref,
            observing_seat=observing_seat,
            team=team_for_seat(observing_seat),
            partner_seat=(observing_seat + 2) % 4,
            dealer=self.dealer,
            forehand=self.forehand,
            phase=self.phase,
            current_actor=self.current_actor,
            own_hand=tuple(self.hands[observing_seat]),
            show_actions=self.show_actions,
            publicly_shown_cards=self.publicly_shown_cards,
            contract=self.contract,
            declarer=self.declarer,
            first_leader=self.first_leader,
            completed_tricks=self.completed_tricks,
            current_trick=self.current_trick,
            team_tricks=self.team_tricks,
            match_score=current_score,
            legal_actions=legal_actions,
            deal_terminal=self.complete,
            match_terminal=self.terminal,
        )

    def _legal_actions_for(self, observing_seat: int) -> tuple[PlayCardAction, ...]:
        if self.complete or self.current_actor != observing_seat:
            return ()
        led_suit = self.current_trick[0].card.suit if self.current_trick else None
        allowed = legal_cards(self.hands[observing_seat], led_suit)
        return tuple(
            PlayCardAction(seat=observing_seat, card=card)
            for card in self.hands[observing_seat]
            if card in allowed
        )

    def apply(self, action: PlayCardAction) -> "FastTrickState":
        """Apply one validated trick-play action without rebuilding engine events."""

        if self.complete:
            raise FastTrickError("fast state is complete")
        if not isinstance(action, PlayCardAction):
            raise FastTrickError("fast trick state accepts card-play actions only")
        if self.current_actor != action.seat:
            raise FastTrickError("fast transition received an action from the wrong seat")
        led_suit = self.current_trick[0].card.suit if self.current_trick else None
        try:
            validate_play(
                hand=self.hands[action.seat], card=action.card, led_suit=led_suit
            )
        except Exception as exc:  # the authoritative rules error is not public here
            raise FastTrickError(f"illegal fast-path play: {exc}") from exc

        mutable_hands = list(self.hands)
        hand = mutable_hands[action.seat]
        card_position = hand.index(action.card)
        mutable_hands[action.seat] = hand[:card_position] + hand[card_position + 1 :]
        hands = tuple(mutable_hands)
        partial = self.current_trick + (PlayedCard(seat=action.seat, card=action.card),)
        if len(partial) < 4:
            return replace(
                self,
                hands=hands,
                current_actor=(action.seat + 1) % 4,
                current_trick=partial,
            )

        winner = trick_winner(partial)
        completed = self.completed_tricks + (
            CompletedTrick(
                trick_index=len(self.completed_tricks),
                leader=partial[0].seat,
                plays=partial,
                winner=winner,
            ),
        )
        totals = list(self.team_tricks)
        totals[team_for_seat(winner).value] += 1
        team_totals = (totals[0], totals[1])
        if len(completed) < 13:
            return replace(
                self,
                hands=hands,
                current_actor=winner,
                current_trick=(),
                completed_tricks=completed,
                team_tricks=team_totals,
            )

        if self.contract is Contract.NOLO:
            deal_score = score_nolo(*team_totals)
        else:
            if self.declarer is None:
                raise FastTrickError("RAMI fast completion lacks a declarer")
            deal_score = score_rami_for_declarer(
                self.declarer, *team_totals
            )
        ending_score = transition_rising_score(self.starting_score, deal_score)
        terminal = is_match_terminal(ending_score)
        if terminal and match_winner(ending_score) is None:
            raise FastTrickError("terminal fast score has no match winner")
        return replace(
            self,
            hands=hands,
            phase=EnginePhase.MATCH_COMPLETE if terminal else EnginePhase.DEAL_COMPLETE,
            current_actor=None,
            current_trick=(),
            completed_tricks=completed,
            team_tricks=team_totals,
            deal_score=deal_score,
            ending_score=ending_score,
            next_dealer=None if terminal else rotate_dealer(self.dealer),
        )

"""Bank of realistic sooli positions (project_target_v2), for evaluating sooli play.

Random deals (random starting score) are shown by the rollout clone. At every
sooli offer whose hand passes the static prefilter, the offer is decided as
the best agent decides it (search, sooli play in the rollouts by the

import argparse
import ctypes as C
import pickle
import random
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

_W: dict = {}


def _init(worlds: int, margin: float) -> None:
    import torch

    torch.set_num_threads(1)
    from tuppi_ai import model as qmodel
    from tuppi_ai import native
    from tuppi_ai.best import EQUITY, ROLLOUT
    from tuppi_ai.equity import Equity

    net = qmodel.load(ROLLOUT, "cpu")
    _W.update(net=net, mlp=native.Mlp(net), worlds=worlds, margin=margin,
              env=native.Env(worlds * 14, 1, Equity.load(EQUITY).table, single_deal=True, encoder=net.encoder),
              feat=np.zeros((1, native.feature_dim(net.encoder)), dtype=np.uint8))


def _greedy(d):
    from tuppi_ai import native
    native.encode(native.observe(d, d.cur), _W["feat"][0], _W["net"].encoder)
    q = _W["mlp"].forward(_W["feat"])[0]
    return max(native.legal(d), key=lambda a: q[a])


def _search(d, actions, play_heur, rng):
    from tuppi_ai import native
    _W["env"].set_sooli(1, False, play_heur)
    return native.search_values(_W["env"], _W["mlp"], native.observe(d, d.cur), actions, _W["worlds"], 6, rng)


def chunk(start: int, count: int) -> list[dict]:
    from tuppi_ai import native
    out = []
    for i in range(start, start + count):
        rng = random.Random(f"sooli-bank|{i}")
        st = C.c_uint64(i * 7919 + 1)
        cards = list(range(52))
        rng.shuffle(cards)
        hands = [sum(1 << c for c in cards[s * 13:(s + 1) * 13]) for s in range(4)]
        level = rng.randrange(13)
        score = (0, 0) if level == 0 else ((4 * level, 0) if rng.random() < 0.5 else (0, 4 * level))
        d = native.new_deal(hands, rng.randrange(4), score[0], score[1], ruleset=2)
        while d.phase == native.PH_SHOW:
            native.lib.tc_apply(d, _greedy(d))
        if d.phase != native.PH_SOOLI_OFFER:
            continue
        for _ in range(2):
            if native.lib.tc_sooli_offer_prefilter(d.hand[d.cur]):
                v = _search(d, [native.ACT_SOOLI_DECLINE, native.ACT_SOOLI_ACCEPT], True, st)
                if v[1] - v[0] > _W["margin"]:
                    offer = bytes(d)
                    native.lib.tc_apply(d, native.ACT_SOOLI_ACCEPT)
                    legal = native.legal(d)
                    h = native.sooli_policy(native.observe(d, d.cur), st)
                    order = [h] + [a for a in legal if a != h]
                    give = order[int(np.argmax(_search(d, order, False, st)))]
                    native.lib.tc_apply(d, give)
                    native.lib.tc_apply(d, native.sooli_policy(native.observe(d, d.cur), st))
                    out.append({"index": i, "offer": offer, "start": bytes(d), "gain": float(v[1] - v[0])})
                    break
            native.lib.tc_apply(d, native.ACT_SOOLI_DECLINE)
    return out


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--deals", type=int, default=20000)
    parser.add_argument("--worlds", type=int, default=64)
    parser.add_argument("--margin", type=float, default=0.02)
    parser.add_argument("--workers", type=int, default=10)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    t0 = time.time()
    size = 200
    starts = list(range(0, args.deals, size))
    with ProcessPoolExecutor(args.workers, initializer=_init, initargs=(args.worlds, args.margin)) as pool:
        rows = [r for part in pool.map(chunk, starts, [size] * len(starts)) for r in part]
    args.out.parent.mkdir(parents=True, exist_ok=True)
    args.out.write_bytes(pickle.dumps(rows))
    print(f"{len(rows)} soolis from {args.deals} deals in {time.time() - t0:.0f} s -> {args.out}")


if __name__ == "__main__":
    main()

def _opts(spec: str) -> tuple[str, dict]:
    name, *rest = spec.split(":")
    return name, dict(p.split("=") for p in rest)


def _init(max_worlds: int) -> None:
    import torch

    torch.set_num_threads(1)
    from tuppi_ai import model as qmodel
    from tuppi_ai import native
    from tuppi_ai.best import EQUITY, ROLLOUT
    from tuppi_ai.equity import Equity

    net = qmodel.load(ROLLOUT, "cpu")
    _W.update(mlp=native.Mlp(net),
              env=native.Env(max_worlds * 14, 1, Equity.load(EQUITY).table, single_deal=True, encoder=net.encoder))


def _move(d, spec, st, rng):
    from tuppi_ai import native
    legal = native.legal(d)
    if len(legal) == 1:
        return legal[0]
    obs = native.observe(d, d.cur)
    h = native.sooli_policy(obs, st)
    name, o = _opts(spec)
    if name == "heur":
        return h
    order = [h] + [a for a in legal if a != h]
    rami = d.cur != d.soolaaja
    if name == "pi":
        v = native.sooli_move_values(d, order)
        return order[int(np.argmin(v) if rami else np.argmax(v))]
    if name == "knows":
        me, other, partner = d.cur, (d.cur + 2) % 4, (d.soolaaja + 2) % 4
        voids = set()
        for t in range(0, d.n_played, 3):
            led = d.play_card[t] // 13
            for i in range(t, min(t + 3, d.n_played)):
                if d.play_seat[i] == other and d.play_card[i] // 13 != led:
                    voids.add(led)
        pool = d.hand[other] | d.hand[partner]
        ok = [c for c in range(52) if pool >> c & 1 and c // 13 not in voids]
        size = bin(d.hand[other]).count("1")
        total = np.zeros(len(order))
        for _ in range(int(o.get("w", 32))):
            m = sum(1 << c for c in rng.sample(ok, size))
            w = native.TcDeal.from_buffer_copy(d)
            w.hand[other], w.hand[partner] = m, pool & ~m
            total += native.sooli_move_values(w, order)
        return order[int(np.argmin(total))]
    env = _W["env"]
    env.set_sooli(int(o.get("b", 1)), False, bool(int(o.get("ph", 0))), float(o.get("l", 0.0)))
    v = native.search_values(env, _W["mlp"], obs, order, int(o.get("w", 64)), 6, st)
    # values are for the observer's team: the soolaaja maximises, and so does the RAMI pair
    return order[int(np.argmax(v))]


def play(args) -> list[int]:
    from tuppi_ai import native
    rows, s_spec, r_spec, seed = args
    out = []
    for row in rows:
        d = native.TcDeal.from_buffer_copy(row["start"])
        st = C.c_uint64(row["index"] * 31 + seed)
        rng = random.Random(row["index"] * 31 + seed)
        while d.phase == native.PH_SOOLI_PLAY:
            spec = s_spec if d.cur == d.soolaaja else r_spec
            native.lib.tc_apply(d, _move(d, spec, st, rng))
        out.append(int(d.sooli_won))
    return out


def evaluate(bank: list, s_spec: str, r_spec: str, workers: int, seed: int = 0) -> list[int]:
    worlds = max(int(_opts(x)[1].get("w", 64)) for x in (s_spec, r_spec))
    parts = [bank[i::workers * 4] for i in range(workers * 4)]
    with ProcessPoolExecutor(workers, initializer=_init, initargs=(worlds,)) as pool:
        res = list(pool.map(play, [(p, s_spec, r_spec, seed) for p in parts]))
    out = [0] * len(bank)
    for i, r in enumerate(res):
        for j, v in enumerate(r):
            out[i + j * workers * 4] = v
    return out


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--bank", type=Path, required=True)
    parser.add_argument("--s", nargs="+", default=["heur"])
    parser.add_argument("--rami", nargs="+", default=["heur"])
    parser.add_argument("--limit", type=int, default=0)
    parser.add_argument("--workers", type=int, default=10)
    parser.add_argument("--out", type=Path)
    args = parser.parse_args()
    bank = pickle.loads(args.bank.read_bytes())
    if args.limit:
        bank = bank[:args.limit]
    results = {}
    for s_spec in args.s:
        for r_spec in args.rami:
            alive = evaluate(bank, s_spec, r_spec, args.workers)
            results[f"{s_spec} | {r_spec}"] = alive
            p = float(np.mean(alive))
            print(f"S {s_spec:28s} RAMI {r_spec:28s} survival {p:.3f} +- {1.96 * (p * (1 - p) / len(alive)) ** 0.5:.3f}",
                  flush=True)
    if args.out:
        args.out.write_text(json.dumps(results))


if __name__ == "__main__":
    main()


=== E032 | AI | tools/gen_game_db.py | complete selected dependency ===
"""Game database: TuppiAI playing itself under the real game, every decision kept.

Each game is recorded completely enough that everything else can be rebuilt
by replaying it through the authoritative engine: the four hands, the dealer,
the starting score, every action, and, for every searched decision, the
search's value of **every candidate action** (equity change for the deciding
seat's team). Those values are free while the game is played and expensive to
recompute, and they are what bidding explanations, drills and value studies
need. Forced moves and heuristic decisions (the sooli return) have no values.

**Shadow bids.** Under locked bids a seat is asked only while every earlier
seat bid NOLO, so the seats after the first RAMI never bid and later seats'
bids would appear only in some deals. After each game, every seat that did
not bid is asked what it would have bid, from exactly the observation it would
have had (bids are blind: its hand, its position and how many bid before it);
`shadow_bids` holds those decisions. They never affect the game.

Output (`--out`): `meta.json` (the bot version, the library and code
identity, the action-id encoding) and one `games_<start>.jsonl.gz` per chunk.
A chunk whose file exists is skipped, so an interrupted run resumes by running
the same command again. A database is one bot version: `meta.json` must match
the current bot or the run refuses to extend it.

    caffeinate -i python tools/gen_game_db.py --deals 5000 --out runs/games/v3_pilot

About 1 deal per second with 3 workers (10.5 CPU-seconds per deal).
"""

from __future__ import annotations

import argparse
import gzip
import hashlib
import json
import random
import subprocess
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

SCHEMA = "tuppi_ai.game_db.v1"
ACTION_IDS = {
    "0-51": "play the card with that index (tuppi_rules.cards.card_to_index)",
    "52-103": "show card (id - 52) declaring NOLO",
    "104-155": "show card (id - 104) declaring RAMI",
    "sooli": "see tuppi_ai.native ACT_SOOLI_DECLINE / ACT_SOOLI_ACCEPT / ACT_EXCHANGE + card",
}


def bot_identity() -> dict:
    from tuppi_ai import best, native
    from tuppi_ai.search import SOOLI_DEFAULTS

    lib = Path(native.__file__).resolve().parents[1] / "native" / "libtuppi.dylib"
    try:
        commit = subprocess.run(["git", "-C", str(ROOT), "rev-parse", "HEAD"], capture_output=True,
                                text=True, check=True).stdout.strip()
        dirty = bool(subprocess.run(["git", "-C", str(ROOT), "status", "--porcelain", "--", "tuppi_ai",
                                     "native"], capture_output=True, text=True).stdout.strip())
    except Exception:
        commit, dirty = "unknown", True
    return {
        "worlds": best.WORLDS, "solver_tail": best.SOLVER_TAIL, "spec": best.SPEC.replace(str(ROOT), "<TuppiAI>"),
        "sooli": SOOLI_DEFAULTS, "libtuppi_sha256": hashlib.sha256(lib.read_bytes()).hexdigest(),
        "tuppiai_commit": commit, "tuppiai_dirty": dirty,
    }


def _score(rng: random.Random) -> tuple[int, int]:
    """0-0 half the time, else a random score the match can start a deal from."""
    if rng.random() < 0.5:
        return (0, 0)
    level = rng.randrange(1, 13)
    return (4 * level, 0) if rng.random() < 0.5 else (0, 4 * level)


def reseed(agents: dict, index: int) -> None:
    """Each game's randomness depends only on its index, not on its chunk."""
    import ctypes as C

    import numpy as np

    for seat, agent in agents.items():
        for searcher in {id(agent.searcher): agent.searcher,
                         id(agent.show_searcher): agent.show_searcher}.values():
            if searcher is None:
                continue
            seed = (index * 4 + seat) * 2654435761 + 12345
            searcher.rng = C.c_uint64(seed & (2**64 - 1))
            searcher.np_rng = np.random.default_rng(seed)
            # The search environment's sooli options as a fresh agent has them:
            # under project_target_v3 they otherwise carry over from the agent's
            # last sooli decision (docs/BIDDING_EXPLANATION_PLAN.md, prerequisite 1).
            searcher.env.set_sooli(0, False, False, 0.0)


def play_chunk(job: tuple[int, int, str, str, int]) -> dict:
    start, count, seed, out, test_every = job
    target = Path(out) / f"games_{start:07d}.jsonl.gz"
    if target.exists():
        return {"start": start, "deals": 0, "skipped": True}
    import torch

    torch.set_num_threads(1)
    from tuppi_ai import native
    from tuppi_ai.best import best_agents
    from tuppi_engine import (ExplicitDeal, action_to_data, apply_match_action, begin_explicit_deal,
                              create_match, match_observation_for)
    from tuppi_engine.state import ACTIVE_PHASES
    from tuppi_engine.trusted import trusted_transitions
    from tuppi_rules import Contract, MatchScore, RulesetId
    from tuppi_rules.cards import FULL_DECK, card_to_code

    native.set_rollout_chunk(128)  # several processes run: least total work, same moves
    agents = best_agents(seed=0)
    games = []
    t0 = time.time()
    with trusted_transitions():
        for index in range(start, start + count):
            rng = random.Random(f"{seed}|{index}")
            cards = list(FULL_DECK)
            rng.shuffle(cards)
            hands = tuple(tuple(cards[s * 13:(s + 1) * 13]) for s in range(4))
            dealer = index % 4
            score = _score(rng)
            match = create_match(initial_score=MatchScore(*score), initial_dealer=dealer,
                                 match_ref=f"db-{index}", ruleset_id=RulesetId.PROJECT_TARGET_V3)
            state = begin_explicit_deal(match, ExplicitDeal(dealer=dealer, hands=hands))
            reseed(agents, index)
            decisions = []
            while state.phase in ACTIVE_PHASES:
                seat = state.current_actor
                observation = match_observation_for(state, seat)
                agent = agents[seat]
                action = agent.choose_action(observation)
                values = getattr(agent, "last_values", None)
                decisions.append({
                    "seat": seat, "phase": observation.phase.value, "action": action_to_data(action),
                    "values": None if values is None else {str(k): round(v, 6) for k, v in values.items()},
                })
                state = apply_match_action(state, action)
            r = state.completed_deals[-1].public_result
            shadow = []
            bid_seats = {d["seat"] for d in decisions if d["phase"] == "SHOWING"}
            if len(bid_seats) < 4:
                reseed(agents, index + 1_000_000_000)  # independent of the game's own draws
                forehand = (dealer + 1) % 4
                for position in range(4):
                    seat = (forehand + position) % 4
                    if seat in bid_seats:
                        continue
                    st = begin_explicit_deal(match, ExplicitDeal(dealer=dealer, hands=hands))
                    for _ in range(position):  # the earlier seats bid NOLO (blind: the card is not seen)
                        earlier = match_observation_for(st, st.current_actor)
                        st = apply_match_action(st, next(a for a in earlier.legal_actions
                                                         if a.declared_contract is Contract.NOLO))
                    if st.current_actor != seat:
                        raise RuntimeError("shadow bid reached the wrong seat")
                    action = agents[seat].choose_action(match_observation_for(st, seat))
                    values = agents[seat].last_values
                    shadow.append({
                        "seat": seat, "position": position, "action": action_to_data(action),
                        "values": None if values is None else {str(k): round(v, 6) for k, v in values.items()},
                    })
            games.append({
                "index": index, "dealer": dealer, "score": list(score), "test": index % test_every == 0,
                "hands": [[card_to_code(c) for c in h] for h in hands], "decisions": decisions,
                "shadow_bids": shadow,
                "result": {"contract": r.contract.value, "declarer": r.declarer, "soolaaja": r.soolaaja,
                           "outcome": r.outcome.value, "points": r.deal_score.points,
                           "winner_team": r.deal_score.winner_team.value, "team_tricks": list(r.team_tricks),
                           "score_after": [r.score_after.team0, r.score_after.team1]},
            })
    tmp = target.with_suffix(".tmp")
    with gzip.open(tmp, "wt") as fh:
        for game in games:
            fh.write(json.dumps(game, separators=(",", ":")) + "\n")
    tmp.rename(target)  # a chunk exists only when complete
    return {"start": start, "deals": count, "skipped": False, "seconds": round(time.time() - t0, 1)}


def load_games(root: Path):
    """Every game in a database, in index order."""
    for path in sorted(root.glob("games_*.jsonl.gz")):
        with gzip.open(path, "rt") as fh:
            for line in fh:
                yield json.loads(line)


def main() -> None:
    p = argparse.ArgumentParser()
    p.add_argument("--deals", type=int, required=True)
    p.add_argument("--out", type=Path, required=True)
    p.add_argument("--seed", default="game-db-v3")
    p.add_argument("--workers", type=int, default=3)
    p.add_argument("--chunk", type=int, default=50)
    p.add_argument("--first", type=int, default=0)
    p.add_argument("--test-every", type=int, default=10, help="mark every n-th deal as held out")
    args = p.parse_args()
    args.out.mkdir(parents=True, exist_ok=True)
    identity = bot_identity()
    meta_path = args.out / "meta.json"
    meta = {"schema": SCHEMA, "ruleset": "project_target_v3", "seed": args.seed, "test_every": args.test_every,
            "chunk": args.chunk, "bot": identity, "action_ids": ACTION_IDS}
    if meta_path.exists():
        old = json.loads(meta_path.read_text())
        old_cmp = {k: v for k, v in old.items() if k != "created"}
        if old_cmp != meta:
            raise SystemExit(f"{meta_path} describes another bot or setup; use a new --out directory")
    else:
        meta["created"] = time.strftime("%Y-%m-%d %H:%M:%S")
        meta_path.write_text(json.dumps(meta, indent=1))
    if identity["tuppiai_dirty"]:
        print("warning: TuppiAI has uncommitted changes in tuppi_ai/ or native/", flush=True)
    jobs = [(s, min(args.chunk, args.first + args.deals - s), args.seed, str(args.out), args.test_every)
            for s in range(args.first, args.first + args.deals, args.chunk)]
    t0 = time.time()
    done = skipped = 0
    with ProcessPoolExecutor(args.workers) as pool:
        for result in pool.map(play_chunk, jobs):
            if result["skipped"]:
                skipped += 1
                continue
            done += result["deals"]
            rate = done / max(time.time() - t0, 1e-9)
            left = sum(j[1] for j in jobs) - done - skipped * args.chunk
            print(f"{done} new deals ({skipped} chunks already there), {rate:.2f} deals/s, "
                  f"about {left / max(rate, 1e-9) / 60:.0f} min left", flush=True)
    print(f"done: {done} new deals in {time.time() - t0:.0f}s")


if __name__ == "__main__":
    main()

"""Behaviour cloning of the incumbent from tools/gen_bc.py shards.

The network is the same QNet; its outputs are trained as logits over legal
actions. The result is a native-speed copy of the incumbent that later serves
as a stable rollout policy for policy improvement.

    python -m tuppi_ai.bc --data runs/bc/v1 --out runs/bc_v1.pt --epochs 3
"""

from __future__ import annotations

import argparse
import math
import time
from pathlib import Path

import numpy as np
import torch

from . import model as qmodel
from .native import NUM_ACTIONS, feature_dim


def load(data: Path):
    feats, acts, masks = [], [], []
    for path in sorted(data.glob("shard_*.npz")):
        z = np.load(path)
        feats.append(z["feat"]); acts.append(z["act"]); masks.append(z["mask"])
    return np.concatenate(feats), np.concatenate(acts).astype(np.int64), np.concatenate(masks)


def batch(feat, mask, idx, dim):
    x = np.unpackbits(feat[idx], axis=1)[:, :dim]
    m = np.unpackbits(mask[idx], axis=1)[:, :NUM_ACTIONS]
    return x, m


def main() -> None:
    p = argparse.ArgumentParser()
    p.add_argument("--data", type=Path, required=True)
    p.add_argument("--out", type=Path, required=True)
    p.add_argument("--epochs", type=float, default=3.0)
    p.add_argument("--batch", type=int, default=8192)
    p.add_argument("--lr", type=float, default=1e-3)
    p.add_argument("--width", type=int, default=768)
    p.add_argument("--blocks", type=int, default=4)
    p.add_argument("--encoder", type=int, default=2)
    args = p.parse_args()
    device = torch.device("mps" if torch.backends.mps.is_available() else "cpu")
    dim = feature_dim(args.encoder)
    feat, act, mask = load(args.data)
    n = len(act)
    rng = np.random.default_rng(0)
    perm = rng.permutation(n)
    hold = perm[: n // 50]
    train = perm[n // 50:]
    print(f"{n} decisions, {len(train)} train, {len(hold)} held out", flush=True)
    net = qmodel.QNet(args.width, args.blocks, args.encoder).to(device)
    opt = torch.optim.AdamW(net.parameters(), lr=args.lr, weight_decay=1e-5, fused=device.type == "mps")
    steps = int(args.epochs * len(train) / args.batch)
    sched = torch.optim.lr_scheduler.LambdaLR(opt, lambda s: 0.5 * (1 + math.cos(math.pi * min(s / steps, 1.0))))
    t0 = time.time()

    def held_out():
        net.eval()
        correct = total = 0
        with torch.no_grad():
            for start in range(0, len(hold), 16384):
                idx = hold[start:start + 16384]
                x, m = batch(feat, mask, idx, dim)
                q = net(torch.from_numpy(x).to(device)).cpu().numpy()
                pick = np.where(m == 1, q, -np.inf).argmax(1)
                correct += int((pick == act[idx]).sum())
                total += len(idx)
        net.train()
        return correct / total

    for step in range(steps + 1):
        idx = train[rng.integers(0, len(train), args.batch)]
        x, m = batch(feat, mask, idx, dim)
        logits = net(torch.from_numpy(x).to(device))
        logits = logits.masked_fill(torch.from_numpy(m).to(device) == 0, -1e9)
        loss = torch.nn.functional.cross_entropy(logits, torch.from_numpy(act[idx]).to(device))
        opt.zero_grad(set_to_none=True)
        loss.backward()
        opt.step()
        sched.step()
        if step % 500 == 0:
            print(f"step {step}/{steps} loss {loss.item():.4f} held-out agreement {held_out():.4f} "
                  f"({time.time() - t0:.0f}s)", flush=True)
    qmodel.save(net, args.out, bc={"data": str(args.data), "held_out_agreement": held_out()})
    print("saved", args.out)


if __name__ == "__main__":
    main()

"""Distil a search policy into a network (expert iteration).

Trains the policy logits by cross-entropy on the actions chosen by
`bsearch` (search over the current blueprint). Rows where no data exists stay
covered by starting from the blueprint's own weights. Optionally mixes in the
original behaviour-cloning shards so rarely searched positions do not drift.

    python -m tuppi_ai.distill --data runs/xi/v1 --init runs/bc_v1.pt --out runs/xi_v1.pt
"""

from __future__ import annotations

import argparse
import math
import time
from pathlib import Path

import numpy as np
import torch

from . import model as qmodel
from .native import NUM_ACTIONS, feature_dim


def load_search(data: Path):
    F, M, A, B, P, V = [], [], [], [], [], []
    for path in sorted(data.glob("shard_*.npz")):
        z = np.load(path)
        F.append(z["feat"]); M.append(z["mask"]); A.append(z["act"]); B.append(z["bp"]); P.append(z["phase"])
        V.append(z["val"].astype(np.float16))
    return (np.concatenate(F), np.concatenate(M), np.concatenate(A).astype(np.int64),
            np.concatenate(B).astype(np.int64), np.concatenate(P), np.concatenate(V))


def main() -> None:
    p = argparse.ArgumentParser()
    p.add_argument("--data", type=Path, required=True, nargs="+")
    p.add_argument("--init", type=Path, required=True)
    p.add_argument("--out", type=Path, required=True)
    p.add_argument("--bc-data", type=Path, help="behaviour-cloning shards to mix in")
    p.add_argument("--bc-share", type=float, default=0.25)
    p.add_argument("--epochs", type=float, default=4.0)
    p.add_argument("--batch", type=int, default=4096)
    p.add_argument("--lr", type=float, default=3e-4)
    p.add_argument("--soft", type=float, default=0.0,
                   help="if > 0, train toward softmax(search values / soft) instead of the searched action")
    args = p.parse_args()
    device = torch.device("mps" if torch.backends.mps.is_available() else "cpu")
    net = qmodel.load(args.init, "cpu")
    net.train().to(device)
    dim = feature_dim(net.encoder)
    parts = [load_search(d) for d in args.data]
    F = np.concatenate([x[0] for x in parts]); M = np.concatenate([x[1] for x in parts])
    A = np.concatenate([x[2] for x in parts]); B = np.concatenate([x[3] for x in parts])
    V = np.concatenate([x[5] for x in parts]) if args.soft > 0 else None
    n = len(A)
    rng = np.random.default_rng(0)
    perm = rng.permutation(n)
    hold, train = perm[: n // 20], perm[n // 20:]
    print(f"{n} search rows; search differs from blueprint on {np.mean(A != B):.3f}", flush=True)
    bc = None
    if args.bc_data:
        from .bc import load as load_bc
        bc = load_bc(args.bc_data)
        print(f"mixing {len(bc[1])} behaviour-cloning rows at share {args.bc_share}", flush=True)
    opt = torch.optim.AdamW(net.parameters(), lr=args.lr, weight_decay=1e-5, fused=device.type == "mps")
    steps = int(args.epochs * len(train) / (args.batch * (1 - (args.bc_share if bc else 0))))
    sched = torch.optim.lr_scheduler.LambdaLR(opt, lambda s: 0.5 * (1 + math.cos(math.pi * min(s / max(steps, 1), 1.0))))

    def unpack(f, m):
        return np.unpackbits(f, axis=1)[:, :dim], np.unpackbits(m, axis=1)[:, :NUM_ACTIONS]

    def held_out():
        net.eval()
        agree_s = agree_b = 0
        with torch.no_grad():
            for s in range(0, len(hold), 16384):
                idx = hold[s:s + 16384]
                x, m = unpack(F[idx], M[idx])
                q = net(torch.from_numpy(x).to(device)).cpu().numpy()
                pick = np.where(m == 1, q, -np.inf).argmax(1)
                agree_s += int((pick == A[idx]).sum())
                agree_b += int((pick == B[idx]).sum())
        net.train()
        return agree_s / len(hold), agree_b / len(hold)

    t0 = time.time()
    print("before: agreement with search %.4f, with blueprint %.4f" % held_out(), flush=True)
    nb = int(args.batch * args.bc_share) if bc else 0
    for step in range(steps + 1):
        idx = train[rng.integers(0, len(train), args.batch - nb)]
        x, m = unpack(F[idx], M[idx])
        y = A[idx]
        if nb:
            j = rng.integers(0, len(bc[1]), nb)
            xb, mb = unpack(bc[0][j], bc[2][j])
            x, m, y = np.concatenate([x, xb]), np.concatenate([m, mb]), np.concatenate([y, bc[1][j]])
        logits = net(torch.from_numpy(x).to(device))
        mt = torch.from_numpy(m).to(device) == 0
        logits = logits.masked_fill(mt, -1e9)
        if V is None:
            loss = torch.nn.functional.cross_entropy(logits, torch.from_numpy(y).to(device))
        else:
            ns = args.batch - nb
            tv = torch.from_numpy(V[idx].astype(np.float32)).to(device) / args.soft
            target = torch.softmax(tv.masked_fill(mt[:ns], -1e9), dim=1)
            logp = torch.log_softmax(logits, dim=1)
            loss = -(target * logp[:ns]).sum(1).mean()
            if nb:
                loss = loss * (ns / args.batch) + torch.nn.functional.cross_entropy(
                    logits[ns:], torch.from_numpy(y[ns:]).to(device)) * (nb / args.batch)
        opt.zero_grad(set_to_none=True)
        loss.backward()
        opt.step()
        sched.step()
        if step % 500 == 0:
            s_, b_ = held_out()
            print(f"step {step}/{steps} loss {loss.item():.4f} held-out agreement: search {s_:.4f} "
                  f"blueprint {b_:.4f} ({time.time() - t0:.0f}s)", flush=True)
    s_, b_ = held_out()
    print(f"final held-out agreement: search {s_:.4f} blueprint {b_:.4f}", flush=True)
    qmodel.save(net, args.out, distill={"data": [str(d) for d in args.data], "init": str(args.init),
                                        "agreement_search": s_, "agreement_blueprint": b_})
    print("saved", args.out, flush=True)


if __name__ == "__main__":
    main()

"""Q-network: observation features -> one value per action id (156).

Outputs 0..51 are card plays and 52..155 are (contract, card) shows. The value
is the acting team's expected match-equity change over the current deal, in
units of `RETURN_SCALE` x equity.
"""

from __future__ import annotations

import torch
from torch import nn

from .native import NUM_ACTIONS, feature_dim

RETURN_SCALE = 10.0


class ResBlock(nn.Module):
    def __init__(self, width: int):
        super().__init__()
        self.norm = nn.LayerNorm(width)
        self.fc1 = nn.Linear(width, width)
        self.fc2 = nn.Linear(width, width)

    def forward(self, x):
        return x + self.fc2(torch.relu(self.fc1(self.norm(x))))


class QNet(nn.Module):
    def __init__(self, width: int = 768, blocks: int = 4, encoder: int = 1):
        super().__init__()
        self.config = {"width": width, "blocks": blocks, "encoder": encoder}
        self.encoder = encoder
        self.inp = nn.Linear(feature_dim(encoder), width)
        self.blocks = nn.Sequential(*[ResBlock(width) for _ in range(blocks)])
        self.norm = nn.LayerNorm(width)
        self.out = nn.Linear(width, NUM_ACTIONS)

    def forward(self, x):
        h = torch.relu(self.inp(x.float()))
        h = self.blocks(h)
        return self.out(torch.relu(self.norm(h)))


def masked_argmax(q: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    return q.masked_fill(mask == 0, float("-inf")).argmax(dim=1)


def save(net: QNet, path, **extra) -> None:
    state = {k: v.detach().cpu() for k, v in net.state_dict().items()}
    torch.save({"config": net.config, "state": state, **extra}, path)


def load(path, device="cpu") -> QNet:
    blob = torch.load(path, map_location=device, weights_only=False)
    net = QNet(**blob["config"])
    net.load_state_dict(blob["state"])
    net.to(device).eval()
    return net

"""Fit the deal-outcome model from incumbent self-play and solve match equity.

    python tools/fit_equity.py --deals 20000 --out runs/equity_incumbent.json
"""

from __future__ import annotations

import argparse
import random
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

from tuppi_ai.equity import DealModel, Equity
from tuppi_ai.incumbent import incumbent_agents

_AGENTS = None


def _play(index: int) -> tuple[bool, int]:
    global _AGENTS
    from tuppi_engine import (EnginePhase, ExplicitDeal, apply_match_action,
                              begin_explicit_deal, create_match, match_observation_for)
    from tuppi_engine.trusted import trusted_transitions
    from tuppi_rules import MatchScore
    from tuppi_rules.cards import FULL_DECK

    if _AGENTS is None:
        _AGENTS = incumbent_agents()
    rng = random.Random(f"fit-equity|{index}")
    cards = list(FULL_DECK)
    rng.shuffle(cards)
    hands = tuple(tuple(cards[s * 13:(s + 1) * 13]) for s in range(4))
    dealer = index % 4
    with trusted_transitions():
        state = begin_explicit_deal(
            create_match(initial_score=MatchScore(0, 0), initial_dealer=dealer, match_ref="fit-equity"),
            ExplicitDeal(dealer=dealer, hands=hands))
        while state.phase in (EnginePhase.SHOWING, EnginePhase.TRICK_PLAY):
            seat = state.current_actor
            state = apply_match_action(state, _AGENTS[seat].choose_action(match_observation_for(state, seat)))
    score = state.completed_deals[-1].public_result.deal_score
    return (score.winner_team.value == dealer % 2), score.points


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--deals", type=int, default=20000)
    parser.add_argument("--workers", type=int, default=12)
    parser.add_argument("--out", type=Path, default=Path("runs/equity_incumbent.json"))
    args = parser.parse_args()
    t0 = time.time()
    with ProcessPoolExecutor(args.workers) as pool:
        outcomes = list(pool.map(_play, range(args.deals), chunksize=64))
    model = DealModel.from_outcomes(outcomes)
    eq = Equity(model)
    args.out.parent.mkdir(parents=True, exist_ok=True)
    eq.save(args.out)
    print(f"{args.deals} deals in {time.time() - t0:.1f}s; dealer_win={model.dealer_win:.4f}")
    print("V(0-0), team 0 deals:", eq.team0(0, 0, 0), " up 48:", eq.team0(48, 0, 0), " down 48:", eq.team0(0, 48, 0))
    try:  # cross-check against TuppiProject's solver on the same model
        from tuppi_valuation.equity import DealModel as VM, EquityTable
        from tuppi_rules import MatchScore, Team
        ref = EquityTable.build(VM(model.dealer_win, model.points_given_dealer, model.points_given_other))
        worst = 0.0
        for s0 in range(0, 52, 4):
            for s1 in ([0] if s0 else range(0, 52, 4)):
                for dealer in range(4):
                    a = eq.team0(s0, s1, dealer)
                    b = ref.equity(MatchScore(s0, s1), Team.TEAM_0, we_deal=(dealer % 2 == 0))
                    worst = max(worst, abs(a - b))
        print(f"max |difference| vs tuppi_valuation.equity: {worst:.2e}")
    except ImportError as exc:
        print("tuppi_valuation unavailable for cross-check:", exc)


if __name__ == "__main__":
    main()

"""The v2 (sooli) world sampler.

Every sampled world must (1) project back to exactly the observer's
observation, (2) be reachable: some exchange and initial deal replay the whole
public history into it, and (3) for a RAMI observer, split the soolaaja's and
partner's shown cards between them with the uniform-exchange-prior odds.
"""

from __future__ import annotations

import ctypes as C
import random

from tuppi_ai import native


def _bits(m):
    return [c for c in range(52) if m >> c & 1]


def _drive(rng, stop_phase, extra_plays=0):
    """A random v2 deal with a sooli, stopped at stop_phase (+ random sooli plays)."""
    while True:
        cards = list(range(52))
        rng.shuffle(cards)
        hands = [sum(1 << c for c in cards[s * 13:(s + 1) * 13]) for s in range(4)]
        deal = native.new_deal(hands, rng.randrange(4), ruleset=2)
        history = []
        while deal.phase not in (native.PH_DONE, native.PH_PLAY) and deal.phase != stop_phase:
            legal = native.legal(deal)
            if deal.phase == native.PH_SHOW:
                # Often NOLO first, so that several seats show before a RAMI.
                rami = [a for a in legal if a >= 104]
                nolo = [a for a in legal if a < 104]
                a = rng.choice(nolo if nolo and rng.random() < 0.6 else (rami or legal))
            else:
                a = rng.choice(legal)
            history.append(a)
            assert native.lib.tc_apply(deal, a) == 0
        if deal.phase != stop_phase:
            continue
        if any(deal.show_verbal[i] for i in range(deal.n_shows)):
            continue  # the samplers ignore what a verbal show says about the hand
        for _ in range(extra_plays):
            if deal.phase != native.PH_SOOLI_PLAY:
                break
            a = rng.choice(native.legal(deal))
            history.append(a)
            native.lib.tc_apply(deal, a)
        if deal.phase == stop_phase:
            return deal, hands, history


def _replay(world, observer_deal, history):
    """Rebuilds initial hands of `world` and replays `history`; returns the end state."""
    S, P = world.soolaaja, (world.soolaaja + 2) % 4
    played = [0, 0, 0, 0]
    for i in range(world.n_played):
        played[world.play_seat[i]] |= 1 << world.play_card[i]
    now = [world.hand[s] | played[s] for s in range(4)]
    phase = observer_deal.phase
    X = next((world.show_card[i] for i in range(world.n_shows) if world.show_seat[i] == S), None)
    Y = next((world.show_card[i] for i in range(world.n_shows) if world.show_seat[i] == P), None)
    if phase == native.PH_SOOLI_RETURN:
        g = observer_deal.given
        init = list(now)
        init[S] |= 1 << g
        init[P] &= ~(1 << g)
        exchange = [native.ACT_EXCHANGE + g]
    else:
        g = X if X is not None and now[P] >> X & 1 else next(c for c in _bits(now[P]) if c != Y)
        r = Y if Y is not None and now[S] >> Y & 1 else next(c for c in _bits(now[S]) if c != X)
        init = list(now)
        init[S] = (now[S] & ~(1 << r)) | (1 << g)
        init[P] = (now[P] & ~(1 << g)) | (1 << r)
        exchange = [native.ACT_EXCHANGE + g, native.ACT_EXCHANGE + r]
    deal = native.new_deal(init, world.dealer, world.score[0], world.score[1], ruleset=2)
    n_pre = world.n_shows + world.n_decisions
    actions = history[:n_pre] + exchange + [world.play_card[i] for i in range(world.n_played)]
    for a in actions:
        if native.lib.tc_apply(deal, a) != 0:
            return None
    return deal


def _check_worlds(deal, history, rng_state, n=30):
    me = deal.cur
    obs = native.observe(deal, me)
    for _ in range(n):
        world = native.sample_world(obs, rng_state)
        assert world is not None
        assert bytes(native.observe(world, me)) == bytes(obs)
        hands = [world.hand[s] for s in range(4)]
        played = sum(1 << world.play_card[i] for i in range(world.n_played))
        assert sum(bin(h).count("1") for h in hands) + bin(played).count("1") == 52
        assert (hands[0] | hands[1] | hands[2] | hands[3] | played) == (1 << 52) - 1
        end = _replay(world, deal, history)
        assert end is not None, "sampled world is not reachable"
        assert [end.hand[s] for s in range(4)] == hands
        assert end.cur == world.cur == deal.cur and end.phase == world.phase

def test_rami_observer_exchange_odds():
    """At the first sooli lead, with both defenders' shown cards unplayed, the
    soolaaja kept its shown card X and the partner kept Y with odds
    (12/13)^2 : 12/13^2 : 12/13^2 : 1/13^2 (uniform exchange prior)."""
    rng = random.Random("v2-sampler-odds")
    state = C.c_uint64(13)
    tried = 0
    while tried < 4:
        deal, _, _ = _drive(rng, native.PH_SOOLI_PLAY)
        S, P = deal.soolaaja, (deal.soolaaja + 2) % 4
        shows = {deal.show_seat[i]: deal.show_card[i] for i in range(deal.n_shows)}
        if S not in shows or P not in shows or deal.cur != deal.declarer:
            continue
        tried += 1
        X, Y = shows[S], shows[P]
        obs = native.observe(deal, deal.cur)
        counts = {}
        n = 20000
        for _ in range(n):
            w = native.sample_world(obs, state)
            key = (w.hand[S] >> X & 1, w.hand[P] >> Y & 1)
            counts[key] = counts.get(key, 0) + 1
        expect = {(1, 1): 144 / 169, (0, 1): 12 / 169, (1, 0): 12 / 169, (0, 0): 1 / 169}
        for key, p in expect.items():
            got = counts.get(key, 0) / n
            sd = (p * (1 - p) / n) ** 0.5
            assert abs(got - p) < 5 * sd + 1e-4, (key, got, p)


def _exposure(hand):
    d = 0
    for v in range(4):
        m = (hand >> (13 * v)) & 0x1FFF
        low, i = ((m & 0xFFF) << 1) | (m >> 12), 0
        for r in range(13):
            if low >> r & 1:
                d += max(0, r - 2 * i)
                i += 1
    return d

    w = native.TcDeal()
    assert native.lib.tc_sample_world_sooli(C.byref(obs), C.byref(state), C.byref(w), mode, lam) == 0
    return w

"""Authoritative showing hierarchy and contract resolution."""

from __future__ import annotations

from collections.abc import Iterable, Sequence

from .cards import ELIGIBLE_SHOW_RANKS, card_color, normalize_cards
from .errors import (
    IncompleteShowingError,
    InvalidCardError,
    InvalidShowActionError,
    InvalidShowingSequenceError,
)
from .model import Card, Color, Contract, Rank, ShowAction, ShowingResolution
from .seating import forehand, validate_seat


def contract_color(contract: Contract) -> Color:
    if not isinstance(contract, Contract):
        raise InvalidShowActionError("declared contract must be a Contract value")
    return Color.RED if contract is Contract.RAMI else Color.BLACK


def is_show_rank_eligible(card_or_rank: Card | Rank) -> bool:
    if isinstance(card_or_rank, Card):
        rank = card_or_rank.rank
    elif isinstance(card_or_rank, Rank):
        rank = card_or_rank
    else:
        raise InvalidCardError("show-rank eligibility requires a Card or Rank value")
    return rank in ELIGIBLE_SHOW_RANKS


def valid_show_cards(
    hand: Iterable[Card], declared_contract: Contract
) -> frozenset[Card]:
    cards = normalize_cards(hand, context="showing hand", allow_empty=False)
    desired_color = contract_color(declared_contract)
    eligible = tuple(card for card in cards if is_show_rank_eligible(card))
    matching = tuple(card for card in eligible if card_color(card) is desired_color)
    if matching:
        return frozenset(matching)
    if eligible:
        return frozenset(eligible)
    return frozenset(cards)


def verbal_declaration_required(
    hand: Iterable[Card], shown_card: Card, declared_contract: Contract
) -> bool:
    cards = normalize_cards(hand, context="showing hand", allow_empty=False)
    if not isinstance(shown_card, Card):
        raise InvalidShowActionError("shown card must be a Card value")
    if shown_card not in cards:
        raise InvalidShowActionError("shown card must belong to the player's hand")
    if shown_card not in valid_show_cards(cards, declared_contract):
        raise InvalidShowActionError(
            "shown card is not allowed by the authoritative fallback hierarchy"
        )
    return not (
        is_show_rank_eligible(shown_card)
        and card_color(shown_card) is contract_color(declared_contract)
    )


def validate_show_action(
    *,
    seat: int,
    hand: Iterable[Card],
    shown_card: Card,
    declared_contract: Contract,
    claimed_verbal_required: bool | None = None,
) -> ShowAction:
    return ShowAction(
        seat=seat,
        hand=hand,
        shown_card=shown_card,
        declared_contract=declared_contract,
        claimed_verbal_required=claimed_verbal_required,
    )


def _derive_show_action_fields(
    *,
    seat: int,
    hand: Iterable[Card],
    shown_card: Card,
    declared_contract: Contract,
    claimed_verbal_required: bool | None,
) -> tuple[int, Card, Contract, bool]:
    validated_seat = validate_seat(seat)
    cards = normalize_cards(hand, context="showing hand", allow_empty=False)
    if len(cards) != 13:
        raise InvalidShowActionError(
            "a show action requires the player's complete 13-card hand"
        )
    verbal_required = verbal_declaration_required(
        cards, shown_card, declared_contract
    )
    if claimed_verbal_required is not None:
        if type(claimed_verbal_required) is not bool:
            raise InvalidShowActionError("claimed verbal_required must be a boolean")
        if claimed_verbal_required is not verbal_required:
            raise InvalidShowActionError(
                "claimed verbal_required disagrees with the value derived from the hand"
            )
    return (validated_seat, shown_card, declared_contract, verbal_required)


def resolve_showing(
    *, dealer: int, actions: Sequence[ShowAction]
) -> ShowingResolution:
    validated_dealer = validate_seat(dealer)
    if isinstance(actions, (str, bytes)):
        raise InvalidShowingSequenceError("actions must be a sequence of ShowAction values")
    try:
        normalized = tuple(actions)
    except TypeError as exc:
        raise InvalidShowingSequenceError(
            "actions must be a sequence of ShowAction values"
        ) from exc
    if len(normalized) > 4:
        raise InvalidShowingSequenceError("showing cannot contain more than four actions")
    for action in normalized:
        if not isinstance(action, ShowAction):
            raise InvalidShowingSequenceError("every showing item must be a ShowAction")
        if (
            type(action.seat) is not int
            or action.seat not in range(4)
            or not isinstance(action.shown_card, Card)
            or not isinstance(action.declared_contract, Contract)
            or type(action.verbal_required) is not bool
        ):
            raise InvalidShowingSequenceError(
                "ShowAction contains invalid or unvalidated fields"
            )
    seats = tuple(action.seat for action in normalized)
    if len(set(seats)) != len(seats):
        raise InvalidShowingSequenceError("a player cannot show more than once")
    expected = tuple((forehand(validated_dealer) + index) % 4 for index in range(4))
    if seats != expected[: len(seats)]:
        raise InvalidShowingSequenceError(
            f"showing seats must follow clockwise order {expected}; received {seats}"
        )

    rami_positions = tuple(
        index
        for index, action in enumerate(normalized)
        if action.declared_contract is Contract.RAMI
    )
    if rami_positions:
        first_rami = rami_positions[0]
        if first_rami != len(normalized) - 1:
            raise InvalidShowingSequenceError(
                "no action is allowed after the first RAMI declaration"
            )
        action = normalized[first_rami]
        return ShowingResolution(
            contract=Contract.RAMI,
            declarer=action.seat,
            actions=normalized,
        )

    if len(normalized) < 4:
        raise IncompleteShowingError(
            "an all-NOLO showing is unresolved until all four players have shown"
        )
    return ShowingResolution(
        contract=Contract.NOLO,
        declarer=None,
        actions=normalized,
    )


def rami_declarer(*, dealer: int, actions: Sequence[ShowAction]) -> int | None:
    return resolve_showing(dealer=dealer, actions=actions).declarer

"""Exact counting over the hard public constraint model.

This module answers questions of the form *"how many hidden-hand assignments does
public information still permit?"* with an exact integer, and *"what fraction of
them place this card at this seat?"* with an exact rational.

Neither answer is a belief. A count is a property of the constraint system, and
the uniform measure it induces is the *combinatorially neutral* one: it treats
every feasible assignment as one object because the hard facts genuinely cannot
distinguish them. It is emphatically **not** a posterior over opponents' hands,
because a real posterior depends on the policies that produced the observed
actions, and this package models no policy. :mod:`tuppi_exact.sampling` states
the same warning where it can actually mislead a search agent.

The method is exact and cheap for one specific structural reason. Two unknown
cards with the same allowed-seat set are interchangeable, so the whole system
collapses to a transportation problem over at most fifteen equivalence classes
and at most three open seats. Counting it is a dynamic program over residual
capacity vectors, with an exact multinomial factor for the cards inside each
class - no permanent, no inclusion-exclusion, and no sampling.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from fractions import Fraction
from functools import lru_cache
import math
from types import MappingProxyType
from typing import Iterator, Mapping

from tuppi_rules import Card

from .constraints import HardConstraintModel
from .errors import CountingError


COUNTING_SCHEMA_VERSION = "tuppi.exact.feasible_assignment_count.v1"


@dataclass(frozen=True, slots=True)
class ClassAllocation:
    """One interchangeable group of unknown cards and the seats it may go to."""

    allowed_seats: tuple[int, ...]
    cards: tuple[Card, ...]

    def __post_init__(self) -> None:
        if type(self.allowed_seats) is not tuple or not self.allowed_seats:
            raise CountingError("a class needs a nonempty allowed-seat tuple")
        if tuple(sorted(set(self.allowed_seats))) != self.allowed_seats:
            raise CountingError("class allowed seats must be sorted and unique")
        if type(self.cards) is not tuple or not self.cards:
            raise CountingError("a class needs at least one card")

    @property
    def size(self) -> int:
        return len(self.cards)


@dataclass(frozen=True, slots=True)
class CountingPlan:
    """The exact transportation problem induced by a constraint model.

    ``seat_order`` fixes a canonical index for every open seat so that residual
    capacity vectors are comparable, hashable, and reproducible. The plan is a
    pure function of the model, so two runs on the same observation produce the
    same plan and therefore the same counts and the same samples.
    """

    seat_order: tuple[int, ...]
    capacities: tuple[int, ...]
    classes: tuple[ClassAllocation, ...]
    schema_version: str = field(default=COUNTING_SCHEMA_VERSION, init=False)

    def __post_init__(self) -> None:
        if len(self.seat_order) != len(self.capacities):
            raise CountingError("capacities must align with the seat order")
        if tuple(sorted(set(self.seat_order))) != self.seat_order:
            raise CountingError("seat order must be sorted and unique")
        if any(value < 0 for value in self.capacities):
            raise CountingError("capacities must be nonnegative")
        total = sum(item.size for item in self.classes)
        if total != sum(self.capacities):
            raise CountingError(
                "class card total disagrees with the total residual capacity"
            )
        for item in self.classes:
            if any(seat not in self.seat_order for seat in item.allowed_seats):
                raise CountingError("a class allows a seat with no residual capacity")

    @classmethod
    def build(cls, model: HardConstraintModel) -> "CountingPlan":
        if not isinstance(model, HardConstraintModel):
            raise CountingError("CountingPlan.build requires a HardConstraintModel")
        seat_order = model.unknown_seats
        classes = tuple(
            ClassAllocation(
                allowed_seats=tuple(sorted(seats)),
                cards=cards,
            )
            for seats, cards in model.equivalence_classes()
        )
        return cls(
            seat_order=seat_order,
            capacities=tuple(model.capacity[seat] for seat in seat_order),
            classes=classes,
        )

    def positions_for(self, allocation: ClassAllocation) -> tuple[int, ...]:
        return tuple(self.seat_order.index(seat) for seat in allocation.allowed_seats)

    def without(self, *, card: Card, seat: int) -> "CountingPlan | None":
        """Return the plan that remains after pinning ``card`` to ``seat``.

        This is the exact reduction used for a marginal count: the pinned card
        leaves its class, and the seat's residual capacity falls by one. The
        other cards of that class stay free, because pinning one member of an
        interchangeable group says nothing about the rest.

        ``None`` means the pinning is infeasible, so its exact count is zero.
        """

        if seat not in self.seat_order:
            raise CountingError("cannot pin a card to a seat with no capacity")
        position = self.seat_order.index(seat)
        if self.capacities[position] == 0:
            raise CountingError("cannot pin a card to a seat with no capacity")
        rebuilt: list[ClassAllocation] = []
        removed = False
        for item in self.classes:
            if not removed and card in item.cards:
                if seat not in item.allowed_seats:
                    raise CountingError("the pinned seat is excluded for this card")
                remaining = tuple(other for other in item.cards if other != card)
                removed = True
                if remaining:
                    rebuilt.append(
                        ClassAllocation(
                            allowed_seats=item.allowed_seats, cards=remaining
                        )
                    )
                continue
            rebuilt.append(item)
        if not removed:
            raise CountingError("the pinned card is not an unknown card of this plan")
        capacities = list(self.capacities)
        capacities[position] -= 1
        surviving_seats = tuple(
            self.seat_order[index]
            for index in range(len(self.seat_order))
            if capacities[index] > 0
        )
        if any(
            not set(item.allowed_seats) & set(surviving_seats) for item in rebuilt
        ):
            # A card whose every allowed seat is now full makes the reduced
            # problem infeasible. Report that as "no plan", so callers can ask
            # about impossible pinnings and receive an exact zero.
            return None
        trimmed = tuple(
            ClassAllocation(
                allowed_seats=tuple(
                    seat for seat in item.allowed_seats if seat in surviving_seats
                ),
                cards=item.cards,
            )
            for item in rebuilt
        )
        return CountingPlan(
            seat_order=surviving_seats,
            capacities=tuple(value for value in capacities if value > 0),
            classes=trimmed,
        )

    def identity_key(self) -> tuple:
        """A hashable canonical key, used for memoization across calls."""

        return (
            self.seat_order,
            self.capacities,
            tuple((item.allowed_seats, item.size) for item in self.classes),
        )


def bounded_compositions(total: int, bounds: tuple[int, ...]) -> Iterator[tuple[int, ...]]:
    """Yield every way to split ``total`` items into parts within ``bounds``."""

    if not bounds:
        if total == 0:
            yield ()
        return
    head, *rest = bounds
    remaining_capacity = sum(rest)
    lowest = max(0, total - remaining_capacity)
    for value in range(lowest, min(head, total) + 1):
        for tail in bounded_compositions(total - value, tuple(rest)):
            yield (value,) + tail


@lru_cache(maxsize=65_536)
def _bounded_compositions_cached(
    total: int, bounds: tuple[int, ...]
) -> tuple[tuple[int, ...], ...]:
    """Return the canonical composition sequence without rebuilding it.

    The generator above remains the reference implementation and preserves its
    public lazy behaviour.  Exact sampling repeatedly asks for the same small
    ``(total, bounds)`` shapes while skipping indexed assignments; caching the
    materialized sequence removes recursion and tuple allocation from that hot
    path without changing ordering or values.
    """

    return tuple(bounded_compositions(total, bounds))


def multinomial(total: int, parts: tuple[int, ...]) -> int:
    ways = 1
    left = total
    for part in parts:
        ways *= math.comb(left, part)
        left -= part
    if left != 0:
        raise CountingError("multinomial parts do not sum to the total")
    return ways


@lru_cache(maxsize=8192)
def _count_from(
    key: tuple, index: int, remaining: tuple[int, ...]
) -> int:
    seat_order, capacities, class_shapes = key
    del capacities
    if index == len(class_shapes):
        return 1 if not any(remaining) else 0
    left = sum(shape[1] for shape in class_shapes[index:])
    if left != sum(remaining):
        return 0
    allowed_seats, size = class_shapes[index]
    positions = tuple(seat_order.index(seat) for seat in allowed_seats)
    bounds = tuple(remaining[position] for position in positions)
    total = 0
    for split in bounded_compositions(size, bounds):
        following = list(remaining)
        for position, value in zip(positions, split):
            following[position] -= value
        deeper = _count_from(key, index + 1, tuple(following))
        if deeper:
            total += multinomial(size, split) * deeper
    return total


def count_plan(plan: CountingPlan) -> int:
    """Return the exact number of assignments the plan permits."""

    if not isinstance(plan, CountingPlan):
        raise CountingError("count_plan requires a CountingPlan")
    if not plan.classes:
        return 1 if not any(plan.capacities) else 0
    return _count_from(plan.identity_key(), 0, plan.capacities)


def feasible_assignment_count(model: HardConstraintModel) -> int:
    """Return the exact number of feasible hidden-hand assignments.

    Zero means the public record admits no assignment at all, which for a legal
    ``project_target_v1`` state cannot happen and indicates a defect in the
    constraint model or a forged observation. One means the hidden hands are
    fully determined by public information.
    """

    if not isinstance(model, HardConstraintModel):
        raise CountingError("feasible_assignment_count requires a constraint model")
    return count_plan(CountingPlan.build(model))


def assignments_with_card_at_seat(
    model: HardConstraintModel, card: Card, seat: int
) -> int:
    """Return how many feasible assignments place ``card`` at ``seat``."""

    if not isinstance(model, HardConstraintModel):
        raise CountingError("assignments_with_card_at_seat needs a constraint model")
    if not isinstance(card, Card):
        raise CountingError("assignments_with_card_at_seat requires a Card")
    if type(seat) is not int or seat not in range(4):
        raise CountingError("seat must be an integer in 0..3")
    if card not in model.allowed_seat_map:
        proven = model.proven_owner(card)
        if proven is None:
            return 0
        return feasible_assignment_count(model) if proven == seat else 0
    if seat not in model.allowed_seats(card):
        return 0
    reduced = CountingPlan.build(model).without(card=card, seat=seat)
    return 0 if reduced is None else count_plan(reduced)


def exact_ownership_counts(
    model: HardConstraintModel,
) -> Mapping[Card, tuple[int, ...]]:
    """Return, per unknown card, how many assignments place it at each seat.

    The four entries always sum to the total feasible count, which the test
    suite checks directly: an off-by-one in the reduction would break that
    identity immediately.
    """

    counts: dict[Card, tuple[int, ...]] = {}
    for card in model.unknown_cards:
        counts[card] = tuple(
            assignments_with_card_at_seat(model, card, seat) for seat in range(4)
        )
    return MappingProxyType(counts)


def exact_uniform_marginals(
    model: HardConstraintModel,
) -> Mapping[Card, tuple[Fraction, ...]]:
    """Return exact rational ownership frequencies under the uniform measure.

    These are exact rationals, not floating-point estimates, and they are a
    property of the constraint system. Reading them as "the chance seat 2 holds
    the ace" imports an assumption this package does not make; see the module
    docstring.
    """

    total = feasible_assignment_count(model)
    if total <= 0:
        raise CountingError("an infeasible model has no uniform measure")
    counts = exact_ownership_counts(model)
    return MappingProxyType(
        {
            card: tuple(Fraction(value, total) for value in per_seat)
            for card, per_seat in counts.items()
        }
    )


def count_summary(model: HardConstraintModel) -> dict[str, object]:
    """Return a compact canonical description of the exact counting result."""

    plan = CountingPlan.build(model)
    total = count_plan(plan)
    return {
        "class_count": len(plan.classes),
        "classes": [
            {"allowed_seats": list(item.allowed_seats), "card_count": item.size}
            for item in plan.classes
        ],
        "feasible_assignment_count": total,
        "feasible_assignment_count_log10": (
            round(math.log10(total), 6) if total > 0 else None
        ),
        "fully_determined": total == 1,
        "open_seat_capacities": [
            {"capacity": capacity, "seat": seat}
            for seat, capacity in zip(plan.seat_order, plan.capacities)
        ],
        "schema_version": COUNTING_SCHEMA_VERSION,
        "structurally_feasible": model.is_structurally_feasible(),
        "unknown_card_count": len(model.unknown_cards),
    }

"""Exactly uniform sampling over the feasible hidden-hand assignments.

The sampler draws from the *uniform* distribution over every assignment the hard
public constraints permit, and it does so exactly: no Metropolis chain, no
accept-reject over the whole assignment, no repair step that quietly biases the
result, and no modulo bias in the underlying integer draws.

The construction is the standard one for a counted structure. The exact counter
already gives, for every prefix of the class decomposition, how many completions
each residual capacity vector admits. Walking the classes and choosing each
split with probability proportional to ``multinomial(split) x completions`` is
therefore uniform over assignments by construction, and choosing which specific
cards fill each part uses unranked combinations, which is uniform over the
multinomial arrangements. Composing two exactly uniform choices leaves the
result exactly uniform.

**What uniform-over-feasible is not.** It is not a posterior over opponents'
hands. A posterior would have to condition on the fact that the opponents chose
the actions they chose, which requires a model of how they choose - and this
package deliberately contains none. Uniform-over-feasible is the neutral
reference measure: correct as a determinization scheme whose bias is stated
rather than hidden, and the right baseline against which a later phase's belief
model must justify itself.

Determinism. Every draw comes from a SHA-256 counter stream seeded by explicit
canonical material. No module-global random state is touched, no clock or
process identity enters, and re-running the same request reproduces the same
assignments in the same order.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from functools import lru_cache
import hashlib
import math
from typing import Mapping

from tuppi_engine import PlayerObservation
from tuppi_rules import Card, card_to_code, card_to_index

from .constraints import HardConstraintModel
from .counting import (
    ClassAllocation,
    CountingPlan,
    _count_from,
    _bounded_compositions_cached,
    multinomial,
    count_plan,
    feasible_assignment_count,
)
from .errors import SamplingError


SAMPLER_SCHEMA_VERSION = "tuppi.exact.uniform_feasible_sampler.v1"
SAMPLER_ALGORITHM_ID = "tuppi.exact.counted_prefix_uniform_v1"
RANDOM_SOURCE_ID = "sha256_counter_bit_reservoir_rejection_v1"


class DeterministicBitSource:
    """An unbiased deterministic source of uniform integers.

    Bits come from ``sha256(seed || counter)`` blocks held in a reservoir.
    :meth:`below` uses rejection on exactly ``bound.bit_length()`` bits, so the
    result is *exactly* uniform on ``range(bound)``. The deal generator's
    documented ``..._modulo_v1`` shortcut is deliberately not reused here:
    modulo bias is negligible when reducing 256 bits to 52, but a sampler whose
    whole claim is exact uniformity must not carry any bias at all.
    """

    __slots__ = ("_seed", "_counter", "_buffer", "_bits", "draws", "rejections")

    def __init__(self, seed: str) -> None:
        if not isinstance(seed, str) or not seed:
            raise SamplingError("the bit source requires a nonempty string seed")
        self._seed = seed.encode("utf-8")
        self._counter = 0
        self._buffer = 0
        self._bits = 0
        self.draws = 0
        self.rejections = 0

    def _refill(self) -> None:
        block = hashlib.sha256(
            self._seed + b"|" + str(self._counter).encode("ascii")
        ).digest()
        self._counter += 1
        self._buffer = (self._buffer << 256) | int.from_bytes(block, "big")
        self._bits += 256

    def _take(self, count: int) -> int:
        if count <= 0:
            return 0
        while self._bits < count:
            self._refill()
        shift = self._bits - count
        value = self._buffer >> shift
        self._buffer &= (1 << shift) - 1
        self._bits = shift
        return value

    def below(self, bound: int) -> int:
        """Return an exactly uniform integer in ``range(bound)``."""

        if type(bound) is not int or bound <= 0:
            raise SamplingError("the exclusive bound must be a positive integer")
        self.draws += 1
        if bound == 1:
            return 0
        width = bound.bit_length()
        while True:
            candidate = self._take(width)
            if candidate < bound:
                return candidate
            self.rejections += 1


@dataclass(frozen=True, slots=True)
class HiddenHandAssignment:
    """One complete feasible assignment of the unknown cards to seats.

    ``hands`` holds every seat's *current* hand: the publicly known cards plus
    the sampled unknown ones. The observing seat's entry is its real hand, which
    the sampler never alters.
    """

    observing_seat: int
    hands: tuple[tuple[Card, ...], ...]
    sampled_owner: Mapping[Card, int]
    sample_index: int
    schema_version: str = field(default=SAMPLER_SCHEMA_VERSION, init=False)

    def __post_init__(self) -> None:
        if type(self.hands) is not tuple or len(self.hands) != 4:
            raise SamplingError("an assignment needs four immutable hands")
        seen: set[Card] = set()
        for hand in self.hands:
            if type(hand) is not tuple:
                raise SamplingError("each assigned hand must be a tuple")
            for card in hand:
                if card in seen:
                    raise SamplingError("an assignment duplicated a card")
                seen.add(card)

    def hand_for(self, seat: int) -> tuple[Card, ...]:
        if type(seat) is not int or seat not in range(4):
            raise SamplingError("seat must be an integer in 0..3")
        return self.hands[seat]

    def content_key(self) -> tuple[tuple[str, ...], ...]:
        """A canonical comparable key, used to test uniformity and determinism."""

        return tuple(
            tuple(sorted(card_to_code(card) for card in hand)) for hand in self.hands
        )


@dataclass(frozen=True, slots=True)
class UniformFeasibleSampler:
    """A deterministic, exactly uniform sampler over feasible assignments."""

    model: HardConstraintModel
    plan: CountingPlan
    seed: str
    total_assignments: int
    schema_version: str = field(default=SAMPLER_SCHEMA_VERSION, init=False)
    algorithm_id: str = field(default=SAMPLER_ALGORITHM_ID, init=False)
    random_source_id: str = field(default=RANDOM_SOURCE_ID, init=False)
    _class_positions: tuple[tuple[int, ...], ...] = field(
        init=False, repr=False, compare=False
    )
    _counting_key: tuple = field(init=False, repr=False, compare=False)
    _suffix_sizes: tuple[int, ...] = field(init=False, repr=False, compare=False)

    @classmethod
    def build(
        cls, model: HardConstraintModel, *, seed: str
    ) -> "UniformFeasibleSampler":
        if not isinstance(model, HardConstraintModel):
            raise SamplingError("the sampler requires a HardConstraintModel")
        if not isinstance(seed, str) or not seed:
            raise SamplingError("the sampler requires a nonempty string seed")
        plan = CountingPlan.build(model)
        total = count_plan(plan)
        if total <= 0:
            raise SamplingError(
                "the hard constraints admit no assignment; nothing can be sampled"
            )
        sampler = cls(model=model, plan=plan, seed=seed, total_assignments=total)
        object.__setattr__(
            sampler,
            "_class_positions",
            tuple(plan.positions_for(item) for item in plan.classes),
        )
        object.__setattr__(sampler, "_counting_key", plan.identity_key())
        suffix_sizes = [0] * (len(plan.classes) + 1)
        for position in range(len(plan.classes) - 1, -1, -1):
            suffix_sizes[position] = suffix_sizes[position + 1] + plan.classes[position].size
        object.__setattr__(sampler, "_suffix_sizes", tuple(suffix_sizes))
        return sampler

    @classmethod
    def for_observation(
        cls, observation: PlayerObservation, *, seed: str
    ) -> "UniformFeasibleSampler":
        if not isinstance(observation, PlayerObservation):
            raise SamplingError("for_observation requires a PlayerObservation")
        return cls.build(HardConstraintModel.build(observation), seed=seed)

    # ---------------------------------------------------------------- sampling

    def sample(self, count: int) -> tuple[HiddenHandAssignment, ...]:
        """Draw ``count`` independent, exactly uniform assignments.

        Draws are independent, so repeats are possible and are not filtered out:
        removing them would make the sample uniform over *distinct* assignments
        instead, which is a different and unstated distribution.
        """

        if type(count) is not int or count <= 0:
            raise SamplingError("sample count must be a positive integer")
        source = DeterministicBitSource(self._stream_seed())
        return tuple(
            self._draw(source, index) for index in range(count)
        )

    def sample_one(self, *, index: int = 0) -> HiddenHandAssignment:
        """Draw the assignment at ``index`` of the deterministic stream."""

        if type(index) is not int or index < 0:
            raise SamplingError("sample index must be a nonnegative integer")
        source = DeterministicBitSource(self._stream_seed())
        # The indexed stream is deliberately unchanged.  We consume exactly
        # the same bounded draws as each preceding assignment, but discard the
        # ownership materialisation.  The old implementation rebuilt all
        # preceding HiddenHandAssignment objects, which made a high observer
        # occurrence needlessly quadratic in the simulation index.
        for prior_index in range(index):
            self._skip_draw(source)
        return self._draw(source, index)

    def _stream_seed(self) -> str:
        from tuppi_eval.serialization import content_digest

        return content_digest(
            {
                "algorithm_id": SAMPLER_ALGORITHM_ID,
                "model": self.model.identity_data(),
                "random_source_id": RANDOM_SOURCE_ID,
                "schema_version": SAMPLER_SCHEMA_VERSION,
                "seed": self.seed,
            }
        )

    def _draw(
        self, source: DeterministicBitSource, index: int
    ) -> HiddenHandAssignment:
        remaining = list(self.plan.capacities)
        owner: dict[Card, int] = {}
        for position, allocation in enumerate(self.plan.classes):
            self._draw_class(
                source=source,
                position=position,
                allocation=allocation,
                remaining=remaining,
                owner=owner,
            )
        if any(remaining):
            raise SamplingError("sampling left residual capacity unfilled")
        hands: list[list[Card]] = [
            list(self.model.known_held[seat]) for seat in range(4)
        ]
        for card, seat in owner.items():
            hands[seat].append(card)
        return HiddenHandAssignment(
            observing_seat=self.model.observing_seat,
            hands=tuple(
                tuple(sorted(hand, key=card_to_index)) for hand in hands
            ),
            sampled_owner=dict(owner),
            sample_index=index,
        )

    def _skip_draw(self, source: DeterministicBitSource) -> None:
        """Consume one assignment from ``source`` without materialising it.

        This is intentionally a structural twin of :meth:`_draw`.  It makes
        the same split and combination-rank calls, in the same order, so the
        next materialised assignment is byte-for-byte the one produced by the
        historical ``sample(index + 1)[index]`` implementation.
        """

        remaining = list(self.plan.capacities)
        for position, allocation in enumerate(self.plan.classes):
            self._skip_draw_class(
                source=source,
                position=position,
                allocation=allocation,
                remaining=remaining,
            )
        if any(remaining):
            raise SamplingError("skipping left residual capacity unfilled")

    def _skip_draw_class(
        self,
        *,
        source: DeterministicBitSource,
        position: int,
        allocation: ClassAllocation,
        remaining: list[int],
    ) -> None:
        positions = self._class_positions[position]
        weights, total = _weighted_split_options(
            self._counting_key, position, tuple(remaining)
        )
        if total <= 0:
            raise SamplingError("no split of an equivalence class can be completed")
        target = source.below(total)
        chosen: tuple[int, ...] | None = None
        for split, weight in weights:
            if target < weight:
                chosen = split
                break
            target -= weight
        if chosen is None:  # pragma: no cover - the weights sum to ``total``
            raise SamplingError("weighted split selection fell through")

        available_count = len(allocation.cards)
        for item, _seat, size in zip(
            positions, allocation.allowed_seats, chosen, strict=True
        ):
            if size:
                # _draw_class un-ranks one of these combinations.  Skipping
                # only needs to consume its rank; the next call must observe
                # exactly the same source state.
                source.below(math.comb(available_count, size))
                available_count -= size
            remaining[item] -= size

    def _draw_class(
        self,
        *,
        source: DeterministicBitSource,
        position: int,
        allocation: ClassAllocation,
        remaining: list[int],
        owner: dict[Card, int],
    ) -> None:
        positions = self._class_positions[position]
        weights, total = _weighted_split_options(
            self._counting_key, position, tuple(remaining)
        )
        if total <= 0:
            raise SamplingError("no split of an equivalence class can be completed")
        target = source.below(total)
        chosen: tuple[int, ...] | None = None
        for split, weight in weights:
            if target < weight:
                chosen = split
                break
            target -= weight
        if chosen is None:  # pragma: no cover - the weights sum to ``total``
            raise SamplingError("weighted split selection fell through")

        available = list(allocation.cards)
        for item, seat, size in zip(
            positions, allocation.allowed_seats, chosen, strict=True
        ):
            if size:
                picked = _unrank_combination(
                    available, size, source.below(math.comb(len(available), size))
                )
                for card in picked:
                    owner[card] = seat
                    available.remove(card)
            remaining[item] -= size

    def _completions(self, position: int, remaining: tuple[int, ...]) -> int:
        """Exact number of ways to finish from ``position`` with ``remaining``."""

        if sum(remaining) != self._suffix_sizes[position]:
            return 0
        # _count_from is the exact memoised recurrence used by count_plan.
        # Starting at the suffix index preserves its result while avoiding a
        # new CountingPlan validation and identity construction for every
        # candidate split.
        return _count_from(self._counting_key, position, remaining)

    # ------------------------------------------------------------------ report

    def provenance(self) -> dict[str, object]:
        """Canonical, hand-free provenance for a sampling request."""

        return {
            "algorithm_id": self.algorithm_id,
            "distribution": "uniform_over_feasible_assignments",
            "feasible_assignment_count": self.total_assignments,
            "is_posterior_over_opponent_hands": False,
            "posterior_caveat": (
                "Uniform over feasible assignments is a neutral reference "
                "measure, not a posterior: a posterior would condition on how "
                "the opponents chose their observed actions, and this package "
                "models no policy."
            ),
            "random_source_id": self.random_source_id,
            "schema_version": self.schema_version,
            "seed": self.seed,
            "stream_seed_digest": self._stream_seed(),
            "unknown_card_count": len(self.model.unknown_cards),
        }


def _unrank_combination(
    available: list[Card], size: int, rank: int
) -> tuple[Card, ...]:
    """Return the ``rank``-th size-``size`` combination in colex-free order.

    Unranking is what makes the choice exactly uniform: every one of the
    ``C(len(available), size)`` combinations has exactly one rank.
    """

    if size < 0 or size > len(available):
        raise SamplingError("combination size is out of range")
    total = math.comb(len(available), size)
    if not 0 <= rank < total:
        raise SamplingError("combination rank is out of range")
    chosen: list[Card] = []
    index = 0
    left = size
    remaining_rank = rank
    while left:
        available_after = len(available) - index - 1
        skip = math.comb(available_after, left - 1)
        if remaining_rank < skip:
            chosen.append(available[index])
            left -= 1
        else:
            remaining_rank -= skip
        index += 1
    return tuple(chosen)


@lru_cache(maxsize=131_072)
def _weighted_split_options(
    counting_key: tuple, position: int, remaining: tuple[int, ...]
) -> tuple[tuple[tuple[tuple[int, ...], int], ...], int]:
    """Return exact weighted class splits for one residual plan state.

    The result depends only on the canonical counting-plan shape and residual
    capacities.  Card identities never enter this cache: cards within one
    equivalence class are interchangeable, while the actual card combination
    rank is still consumed by the caller using the original source.
    """

    seat_order, _capacities, class_shapes = counting_key
    allowed_seats, size = class_shapes[position]
    positions = tuple(seat_order.index(seat) for seat in allowed_seats)
    bounds = tuple(remaining[item] for item in positions)
    weights: list[tuple[tuple[int, ...], int]] = []
    total = 0
    for split in _bounded_compositions_cached(size, bounds):
        following = list(remaining)
        for item, value in zip(positions, split):
            following[item] -= value
        completions = _count_from(counting_key, position + 1, tuple(following))
        if not completions:
            continue
        weight = multinomial(size, split) * completions
        total += weight
        weights.append((split, weight))
    return tuple(weights), total


def sample_assignments(
    observation: PlayerObservation, *, seed: str, count: int
) -> tuple[HiddenHandAssignment, ...]:
    """Convenience entry point: build the model and draw ``count`` assignments."""

    sampler = UniformFeasibleSampler.for_observation(observation, seed=seed)
    return sampler.sample(count)


def uniform_sampling_summary(
    observation: PlayerObservation, *, seed: str, count: int
) -> dict[str, object]:
    """Return a canonical summary of a sampling request without leaking a hand."""

    model = HardConstraintModel.build(observation)
    sampler = UniformFeasibleSampler.build(model, seed=seed)

    return {
        "distinct_assignments_drawn": distinct,
        "feasible_assignment_count": feasible_assignment_count(model),
        "provenance": sampler.provenance(),
        "requested_count": count,
        "schema_version": SAMPLER_SCHEMA_VERSION,
    }

def validate_real_game_deal_analysis(
    value: Mapping[str, object],
    *,
    runner: LiveMatchRunner,
    seat: int,
    deal_number: int,
) -> dict[str, object]:
    """Check a cached envelope against the authoritative deal.

    The bot transcript is re-applied through the engine rather than recomputed,
    so reopening an analysis does not run the bots again.
    """

    if not isinstance(value, Mapping) or set(value) != _RESPONSE_KEYS:
        raise RealGameAnalysisError("real-game analysis fields are invalid")
    if value["schema_version"] != REAL_GAME_DEAL_ANALYSIS_SCHEMA_VERSION:
        raise RealGameAnalysisError("real-game analysis uses another schema")
    research = _completed_research_deal(runner, deal_number, seat)
    actual = research.public_result
    expected = {
        "match_ref": runner.match_ref,
        "deal_ref": actual.deal_ref,
        "deal_number": deal_number,
        "seat": seat,
        "ruleset_id": runner.ruleset_id.value,
        "deal_id": research.deal_id,
        "local_team": team_for_seat(seat).value,
        "initial_hands": _hands_data(research),
        "actual_result": deal_result_to_data(actual),
        "actual_exchange": _exchange_data(research.sooli_given, research.sooli_returned),
        "bot_policy": bot_policy_for(runner),
    }
    for key, wanted in expected.items():
        if value[key] != wanted:
            raise RealGameAnalysisError(f"real-game analysis {key} is stale")
    raw_actions = value["bot_actions"]
    if type(raw_actions) is not list or len(raw_actions) > _MAX_REPLAY_ACTIONS:
        raise RealGameAnalysisError("real-game bot transcript is invalid")
    state = _initial_state(research, runner)
    try:
        for raw in raw_actions:
            state = apply_action(state, action_from_data(raw))
    except Exception as exc:  # any engine rejection means the cache is unusable
        raise RealGameAnalysisError("real-game bot transcript is not legal") from exc
    if not state.complete:
        raise RealGameAnalysisError("real-game bot transcript does not finish the deal")
    bot = deal_result_from_state(state)
    review = value["review"]
    if review is not None and not isinstance(review, Mapping):
        raise RealGameAnalysisError("real-game analysis review is invalid")
    if (
        value["bot_replay"] != deal_result_to_data(bot)
        or value["bot_exchange"] != _exchange_data(state.sooli_given, state.sooli_returned)
        or value["comparison"] != _comparison(actual, bot, seat=seat)
        or value["summary"] != _summary(actual, review)
        or value["analyses"] != []
    ):
        raise RealGameAnalysisError("real-game analysis disagrees with its transcript")
    return dict(value)


def _completed_research_deal(
    runner: LiveMatchRunner, deal_number: int, seat: int
) -> ResearchDealResult:
    if not isinstance(runner, LiveMatchRunner):
        raise RealGameAnalysisError("real-game analysis requires a live match runner")
    if runner.ruleset_id not in REAL_GAME_RULESETS:
        raise RealGameAnalysisError("the basic rules use the basic-rules analysis")
    if type(seat) is not int or seat not in range(4):
        raise RealGameAnalysisError("analysis seat must be an integer in 0..3")
    if (
        type(deal_number) is not int
        or deal_number < 0
        or deal_number >= len(runner.state.completed_deals)
    ):
        raise RealGameAnalysisError("deal is not complete")
    return runner.state.completed_deals[deal_number]


def _initial_state(research: ResearchDealResult, runner: LiveMatchRunner):
    actual = research.public_result
    explicit = ExplicitDeal(
        dealer=actual.dealer,
        hands=research.initial_hands,
        claimed_deal_id=research.deal_id,
    )
    return create_deal_state(
        explicit,
        starting_score=actual.score_before,
        match_ref=actual.match_ref,
        deal_ref=actual.deal_ref,
        deal_number=actual.deal_number,
        event_start_index=actual.events[0].event_index,
        ruleset_id=runner.ruleset_id,
    )


def _hands_data(research: ResearchDealResult) -> list[list[str]]:
    return [[card_to_code(card) for card in hand] for hand in research.initial_hands]


def _exchange_data(given, returned) -> dict[str, str] | None:
    if given is None and returned is None:
        return None
    if given is None or returned is None:
        raise RealGameAnalysisError("a completed sooli exchange has both cards")
    return {"given": card_to_code(given), "returned": card_to_code(returned)}


def _summary(actual: DealResult, review: Mapping[str, object] | None) -> dict[str, object]:
    findings = review.get("findings") if review is not None else None
    return {
        "contract": actual.contract.value,
        "diagnostic_count": len(findings) if isinstance(findings, list) else 0,
        "outcome": actual.outcome.value,
        "points": actual.deal_score.points,
        "score_before": [actual.score_before.team0, actual.score_before.team1],
        "score_after": [actual.score_after.team0, actual.score_after.team1],
        "team_tricks": list(actual.team_tricks),
    }
